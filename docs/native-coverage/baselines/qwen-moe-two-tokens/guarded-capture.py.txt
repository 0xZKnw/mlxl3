def tensors(path,tokens,vocab,states):
 rows=[]
 with path.open() as f:
  for line in f:
   j=json.loads(line); logits=j['logits']
   assert len(logits)==vocab and all(type(x) is int and 0<=x<=65535 and (x&0x7c00)!=0x7c00 for x in logits),'logit count/finitude'
   data=array.array('H',logits);assert sys.byteorder=='little' and data.itemsize==2
   snap={}
   for st in j['states']:
    shape=st['shape']; blob=bytes(st['data'])
    assert shape and all(type(n) is int and n>0 for n in shape) and blob and len(blob)%math.prod(shape)==0 and st['name'] not in snap
    snap[st['name']]={'shape':shape,'bytes':len(blob),'sha256':hashlib.sha256(blob).hexdigest()}
   if states:assert snap,'empty states'
   rows.append({'token':j['token'],'logits':len(logits),'logits_sha256':hashlib.sha256(data).hexdigest(),'states':snap})
 assert [x['token'] for x in rows]==tokens
 return rows
