def sha(path):
    with Path(path).open('rb') as handle:
        return hashlib.file_digest(handle, 'sha256').hexdigest()


def save():
    report['elapsed_seconds'] = time.monotonic() - started
    pending = OUTPUT.with_suffix('.tmp')
    pending.write_text(json.dumps(report, indent=2, allow_nan=False) + '\n')

def load_report(path):
    path = ROOT / path
    item = json.loads(path.read_text())
    if item.get('status') != 'complete':
        raise ValueError('Incomplete provenance: ' + str(path))
    report['provenance'].setdefault('reports', {})[str(path)] = sha(path)
    return item


def checked_array(path, expected):
    if sha(path) != expected['sha256'] or path.stat().st_size != expected['bytes']:
        raise ValueError('Array SHA/bytes mismatch: ' + str(path))
    values = np.load(path, mmap_mode='r', allow_pickle=False)
    if values.shape != (1024, 128000) or str(values.dtype) != expected['dtype']:
        raise ValueError('Array shape/dtype mismatch: ' + str(path))
    for first in range(0, 1024, 32):
        if not np.isfinite(values[first:first+32]).all():
            raise ValueError('Nonfinite array: ' + str(path))
    return values, {'path': str(path), 'sha256': expected['sha256'], 'bytes': expected['bytes'],
                    'shape': [1024, 128000], 'dtype': str(values.dtype), 'finite_values_checked': values.size, 'sha_verified': True}


def checked_routes(path, expected):
    if sha(path) != expected['sha256'] or path.stat().st_size != expected['bytes']:
        raise ValueError('Route SHA/bytes mismatch: ' + str(path))
    values = load_file(path)
    if set(values) != {f'{key}.{kind}' for key in LAYERS for kind in ('ids','weights')}:
        raise ValueError('Missing route tensor or unexpected layer')
    result = {}
    for key in LAYERS:
        ids, weights = values[key+'.ids'], values[key+'.weights']
        if ids.shape != (1024,4) or weights.shape != (1024,4) or ids.dtype != np.int64 or weights.dtype != np.float16:
            raise ValueError('Wrong route shape/dtype')
        if not np.isfinite(weights).all() or not ((ids >= 0) & (ids < 64)).all() or not (weights > 0).all():
            raise ValueError('Nonfinite/out-of-range route')
        if not (np.diff(np.sort(ids, axis=1), axis=1) > 0).all() or not (np.abs(weights.astype(np.float64).sum(1)-1) <= 0.002).all():
            raise ValueError('Repeated experts or invalid affinity mass')
        result[key] = (ids, weights)
    return result, {'path':str(path), 'sha256':expected['sha256'], 'bytes':expected['bytes'], 'layers':38, 'tokens_per_layer':1024, 'top_k':4, 'values_checked':38*1024*4*2, 'sha_verified':True}


def bitexact(a,b):
    if a.shape != b.shape or a.dtype != b.dtype:
        return False
    return all(np.array_equal(a[first:first+32].view(np.uint16), b[first:first+32].view(np.uint16)) for first in range(0,len(a),32))


def score(reference, actual, labels):
    total, segment = np.zeros(3,dtype=np.float64), np.zeros(3,dtype=np.float64)
    matches, segment_matches, segments = 0,0,[]
    for index,label in enumerate(labels):
        item = np.asarray(losses(actual[index], reference[index], label),dtype=np.float64)
        if item.shape != (3,) or not np.isfinite(item).all() or item[2] < -1e-9:
            raise ValueError('Invalid F64 scorer output')
        total += item; segment += item
        same = int(np.argmax(actual[index]) == np.argmax(reference[index]))
        matches += same; segment_matches += same
        if (index+1)%128 == 0:
            mean=segment/128
            segments.append({'first':index-127,'last':index,'positions':128,'actual_nll':float(mean[0]),'reference_nll':float(mean[1]),'mean_kl_reference_to_actual':float(mean[2]),'top1_matches':segment_matches})
            segment.fill(0);segment_matches=0
    mean=total/1024
    return {'positions':1024,'vocabulary':128000,'actual_nll':float(mean[0]),'reference_nll':float(mean[1]),'actual_ppl':math.exp(mean[0]),'reference_ppl':math.exp(mean[1]),'mean_kl_reference_to_actual':float(mean[2]),'top1_matches':matches,'top1_fraction':matches/1024,'segments':segments}
