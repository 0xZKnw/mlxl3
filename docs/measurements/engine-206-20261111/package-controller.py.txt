def guard(owned=()):
    heavy = {
        "cargo",
        "rustc",
        "cbmc",
        "cargo-kani",
        "kani",
        "mlxl3-rs",
        "mlxl3",
        "mlxl3-hybrid",
        "response-check",
        "mlxl3-real-release-check",
    }
    for line in subprocess.check_output(["ps", "-axo", "pid=,comm="], text=True).splitlines():
        pid, command = line.strip().split(None, 1)
        name = command.rsplit("/", 1)[-1]
        engine = name.endswith(("mlxl3-rs", "-libtest", "-bintest")) or name.startswith(
            ("mlxl3_native-", "mlxl3_rs-")
        )
        if int(pid) not in owned and (name in heavy or engine):
            raise RuntimeError(f"external heavy job: {pid}/{name}")


def arm_environment(arm, home=None):
    env = os.environ.copy()
    if home is not None:
        env["MLXL3_HOME"] = home
    env["MLXL3_LFM3_DECODE_PIPELINE"] = "0"
    enabled = "1" if arm == "default" else "0"
    env["MLXL3_LFM3_FUSED_HADAMARD"] = "0"
    env["MLXL3_LFM3_KV_SLAB"] = "0"
    for name in ["MLXL3_LFM3_PREFILL_GEMM", "MLXL3_LFM3_PREFILL_GROUPED"]:
        if enabled == "1":
            env.pop(name, None)
        else:
            env[name] = "0"
    env["DYLD_LIBRARY_PATH"] = str(B / arm)
    return env


def wait_quiet():
    deadline = time.monotonic() + 600
    since = None
    while time.monotonic() < deadline:
        try:
            guard()
            since = since or time.monotonic()
            if time.monotonic() - since >= 5:
                return
        except RuntimeError as error:
            since = None
            if not REPORT["waiting"] or REPORT["waiting"][-1]["error"] != str(error):
                REPORT["waiting"].append({"epoch": time.time(), "error": str(error)})
                save()
        time.sleep(1)
    raise TimeoutError("no quiet resource window")


def validate(response, case):
    stats = response["stats"]
    assert response["text"] and len(response["token_hash"]) == 16
    count = stats["generated_tokens"]
    for key in ["client_seconds", "client_first_delta_seconds"]:
        assert (
            isinstance(response[key], (float, int))
            and math.isfinite(response[key])
            and response[key] > 0
        )
    assert response["client_first_delta_seconds"] <= response["client_seconds"]
    assert 0 < count <= case["budget"]
    assert stats["context_limit"] == 4096
    assert stats["cached_prompt_tokens"] == 0
    assert stats["evaluated_prompt_tokens"] == stats["prompt_tokens"] > 0
    assert stats["decode_tokens"] == count - 1
    for key in ["prefill_seconds", "ttft_seconds", "elapsed_seconds"]:
        assert math.isfinite(stats[key]) and stats[key] > 0
    assert math.isfinite(stats["decode_tps"]) and stats["decode_tps"] >= 0
    if "hash" in case:
        assert response["token_hash"] == case["hash"] and count == case.get(
            "eos_count", case["budget"]
        )
        assert stats["prompt_tokens"] == case["prompt_tokens"]
        if "eos_count" in case:
            assert count == case["eos_count"]
            assert response["stop_reason"] == "eos"
    else:
        assert stats["prompt_tokens"] > 0
        if case["name"] in {"hello", "sum"}:
            assert 2 <= stats["prompt_tokens"] < 24


def generate(child, case, label, *, cancel=False, image=False):
    deadline = time.monotonic() + 120
    message = {"role": "user", "content": case["prompt"]}
    if image:
        message["images"] = ["data:image/png;base64,AAAA"]
    request_started = time.perf_counter()
    first_delta = None
    first_answer = None
    child.send(
        {
            "type": "generate",
            "request_id": label,
            "conversation_id": label,
            "messages": case.get("messages", [message]),
            "max_tokens": case["budget"],
            "temperature": 0.0,
            "top_k": 0,
            "repetition_penalty": 1.0,
            "reuse_prompt_cache": False,
            "mtp": False,
            "dflash2": False,
            "memory_saver": False,
        },
        deadline,
    )
    text = ""
    last_guard = time.monotonic()
    cancel_sent = False
    while True:
        if time.monotonic() - last_guard > 1:
            guard([child.process.pid])
            last_guard = time.monotonic()
        event = child.receive(deadline)
        if event.get("request_id") != label:
            continue
        kind = event.get("type")
        if (
            cancel
            and kind == ("delta" if cancel == "decode" else "context_usage")
            and not cancel_sent
        ):
            os.kill(child.process.pid, signal.SIGUSR1)
            cancel_sent = True
        if kind == "delta":
            text += event.get("text", "")
            if event.get("text"):
                first_delta = first_delta or time.perf_counter() - request_started
                if event.get("phase") == "answer":
                    first_answer = first_answer or time.perf_counter() - request_started
        elif kind == "cancelled":
            assert cancel and cancel_sent
            return {"cancelled": True}
        elif kind == "error":
            if image:
                assert "does not support images" in event["message"]
                return {"rejected": event["message"]}
            raise RuntimeError(event)
        elif kind == "complete":
            assert not cancel and not image
            guard([child.process.pid])
            result = {
                "stats": event["stats"],
                "token_hash": event["token_hash"],
                "text": text,
                "stop_reason": event.get("stop_reason"),
                "client_seconds": time.perf_counter() - request_started,
                "client_first_delta_seconds": first_delta,
                "client_first_answer_seconds": first_answer,
            }
            validate(result, case)
            return result


def same(actual, expected):
    for key in ["token_hash", "text", "stop_reason"]:
        assert actual[key] == expected[key], f"generation divergence: {key}"
    for key in ["generated_tokens", "prompt_tokens"]:
        assert actual["stats"][key] == expected["stats"][key], key

def f16_logits(data):
    values = data["logits"]
    assert len(values) == 128000 and all(isinstance(v, int) and 0 <= v <= 65535 for v in values)
    values = np.asarray(values, dtype=np.uint16).view(np.float16).astype(np.float64)
    assert np.isfinite(values).all()
    return values


def kl(actual, expected):
    assert len(actual) == len(expected) == 128000
    assert np.isfinite(actual).all() and np.isfinite(expected).all()
    za = actual.max() + np.log(np.exp(actual - actual.max()).sum())
    ze = expected.max() + np.log(np.exp(expected - expected.max()).sum())
    result = float(np.sum(np.exp(expected - ze) * (expected - ze - actual + za)))
    assert math.isfinite(result) and result <= 0.001, result
    return result


def cli_quality():
    case = json.loads((OLDUP / "installed8-reference.json").read_text())["cases"][0]
    results = {}
    REPORT["cli"] = []
    for length in [5, 128, 256, 499]:
        for arm in ["off", "default"]:
            guard()
            label = f"cli-{length}-{arm}"
            command = [
                str(B / arm / "mlxl3"),
                "forward",
                MODEL,
                "--tokens",
                ",".join(map(str, case["tokens"][:length])),
                "--batch",
            ]
            if length in [5, 256]:
                command.append("--states")
            with (
                (D / (label + ".json")).open("w") as out,
                (D / (label + ".stderr")).open("w") as err,
            ):
                child = subprocess.Popen(
                    command,
                    stdout=out,
                    stderr=err,
                    env=arm_environment(arm),
                    start_new_session=True,
                )
                try:
                    deadline = time.monotonic() + 120
                    while child.poll() is None:
                        guard([child.pid])
                        if time.monotonic() >= deadline:
                            raise TimeoutError("CLI forward timed out")
                        time.sleep(1)
                    assert child.returncode == 0
                finally:
                    if child.poll() is None:
                        os.killpg(child.pid, signal.SIGTERM)
                        try:
                            child.wait(timeout=5)
                        except subprocess.TimeoutExpired:
                            os.killpg(child.pid, signal.SIGKILL)
                            child.wait(timeout=5)
            data = json.loads((D / (label + ".json")).read_text())
            logits = f16_logits(data)
            record = {
                "arm": arm,
                "length": length,
                "command": command,
                "reaped": child.poll() is not None,
            }
            if length == 499:
                record["independent_ExLlama_KL"] = kl(
                    logits, np.asarray(case["logits"], dtype=np.float64)
                )
            if arm == "off":
                results[length] = (data, logits)
            else:
                gold, gold_logits = results[length]
                record["baseline_KL"] = kl(logits, gold_logits)
                if length in [5, 256]:
                    assert len(data["states"]) == len(gold["states"]) == 35
                    errors = []
                    for actual, expected in zip(data["states"], gold["states"]):
                        assert (actual["name"], actual["shape"]) == (
                            expected["name"],
                            expected["shape"],
                        )
                        av = np.frombuffer(bytes(actual["data"]), dtype=np.float16).astype(
                            np.float64
                        )
                        ev = np.frombuffer(bytes(expected["data"]), dtype=np.float16).astype(
                            np.float64
                        )
                        assert len(av) == len(ev) == math.prod(actual["shape"]) > 0
                        assert np.isfinite(av).all() and np.isfinite(ev).all()
                        error = float(np.linalg.norm(av - ev) / max(np.linalg.norm(ev), 1e-30))
                        assert error <= 0.01
                        errors.append(error)
                    record["max_state_relative_L2"] = max(errors)
            REPORT["cli"].append(record)
            save()
