#!/bin/zsh
set -euo pipefail
out="$PWD/build/desktop-followup-20261102"
binary_dir="$PWD/apps/MLXL3Studio/.build/debug"
sources=(apps/MLXL3Studio/Sources/MLXL3Studio/*.swift)
sources=("${(@)sources:#*/MLXL3StudioApp.swift}")
sources=("${(@)sources:#*/StudioModel.swift}")
python3 - <<'PY'
from pathlib import Path
s=Path('apps/MLXL3Studio/Sources/MLXL3Studio/StudioModel.swift').read_text()
a=s.index('        if name == selectedModelName, bridge.isRunning {')
b=s.index('        selectedModelName = name',a)
s=s[:a]+'        if name == selectedModelName, bridge.isRunning { return }\n'+s[b:]
s+='\n'+Path('tests/completion-notifications-check.swift').read_text()
Path('build/desktop-followup-20261102/OldRetryCombined.swift').write_text(s)
PY
swiftc -swift-version 6 -parse-as-library -warnings-as-errors -sdk /Library/Developer/CommandLineTools/SDKs/MacOSX26.5.sdk -module-cache-path "${out}/mutation-modules" -I "${binary_dir}" "${sources[@]}" "${binary_dir}/SwiftMath.o" "${out}/OldRetryCombined.swift" -o "${out}/old-retry-check"
ulimit -c 0
if "${out}/old-retry-check" "$PWD/tests/fake-desktop-engine.py" > "${out}/old-retry-result.log" 2>&1; then
    print -u2 'Old same-model guard unexpectedly passed'
    exit 1
fi
grep -q 'same-model retry' "${out}/old-retry-result.log"
print 'Original same-model retry bug detected in isolated production source copy.'
