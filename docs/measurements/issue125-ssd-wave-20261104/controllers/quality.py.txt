def selfchecks():
    good = {'parity':True,'compared_state_words':115212288,'compared_logits':1986560,'initial_archive_states':84,'context':4096,'cold_control_and_candidate':True,'explicit_legacy_continuation':True,'actual_extra_submissions':4,'steps':[{'token':t,'control_extra_submissions':0,'snapshot_position':i,'actual_extra_submissions':1} for i,t in enumerate([248044,1,42,7],1)]}
    comp={'cases':4,'finite_words':10240,'actual_extra_submissions':2,'max_commands':4}
    model(good); component(comp)
    rejected=0
    bads=[]
    for value, check in [(good,model),(comp,component)]:
        for key in value:
            bad=copy.deepcopy(value);bad.pop(key);bads.append((check,bad))
    for key,value in [('actual_extra_submissions',0),('actual_extra_submissions',True),('steps',[]),('steps',good['steps'][:3]),('steps',good['steps']*2),('parity',False),('compared_state_words',0)]:
        bad=copy.deepcopy(good);bad[key]=value;bads.append((model,bad))
    for key,value in [('token',True),('actual_extra_submissions',-1),('actual_extra_submissions',49),('actual_extra_submissions',True),('control_extra_submissions',1),('snapshot_position',0)]:
        bad=copy.deepcopy(good);bad['steps'][0][key]=value;bads.append((model,bad))
    for key,value in [('cases',True),('finite_words',0),('actual_extra_submissions',0),('max_commands',3)]:
        bad=copy.deepcopy(comp);bad[key]=value;bads.append((component,bad))
    for check,bad in bads:
        try: check(bad)
        except (AssertionError,KeyError,TypeError): rejected+=1
        else: raise AssertionError('invalid evidence accepted')
    assert rejected == len(bads)
    return {'valid':2,'invalid_rejected':rejected,'unchanged_helpers':'P13 final executed helpers; archived39+15 selfchecks'}
