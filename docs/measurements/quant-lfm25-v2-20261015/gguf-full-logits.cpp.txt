#include "llama.h"
#include <cmath>
#include <cstdio>
#include <filesystem>
#include <fstream>
#include <memory>
#include <stdexcept>
#include <string>
#include <vector>

static void require(bool condition, const char *message) {
    if (!condition) throw std::runtime_error(message);
}

int main(int argc, char **argv) {
    try {
        require(argc == 5, "usage: full_logits MODEL IDS OUTDIR BACKEND_DIR");
        std::ifstream ids_file(argv[2]);
        require(ids_file.good(), "cannot open token IDs");
        std::vector<llama_token> ids;
        int64_t value;
        while (ids_file >> value) {
            require(value >= 0 && value < 128000, "token outside LFM vocabulary");
            require(ids.size() < 8192, "too many token IDs");
            ids.push_back(static_cast<llama_token>(value));
        }
        require(ids_file.eof() && ids.size() == 8192, "expected exactly 8192 valid token IDs");
        require(!std::filesystem::exists(argv[3]), "output exists; preserve earlier attempt");
        require(std::filesystem::create_directory(argv[3]), "cannot create output directory");
        ggml_backend_load_all_from_path(argv[4]);
        llama_backend_init();
        const auto backend = std::unique_ptr<int, void (*)(int *)>(
            reinterpret_cast<int *>(1), [](int *) { llama_backend_free(); });
        auto mp = llama_model_default_params();
        mp.n_gpu_layers = 99;
        const auto model = std::unique_ptr<llama_model, decltype(&llama_model_free)>(
            llama_model_load_from_file(argv[1], mp), llama_model_free);
        require(model != nullptr, "model load failed");
        require(llama_vocab_n_tokens(llama_model_get_vocab(model.get())) == 128000,
                "wrong vocabulary size");
        auto cp = llama_context_default_params();
        cp.n_ctx = 1024; cp.n_batch = 128; cp.n_ubatch = 128; cp.n_seq_max = 1;
        cp.type_k = GGML_TYPE_F16; cp.type_v = GGML_TYPE_F16;
        cp.n_threads = 8; cp.n_threads_batch = 8;
        const auto ctx = std::unique_ptr<llama_context, decltype(&llama_free)>(
            llama_init_from_model(model.get(), cp), llama_free);
        require(ctx != nullptr, "context load failed");
        struct OwnedBatch {
            llama_batch batch = llama_batch_init(128, 0, 1);
            ~OwnedBatch() { llama_batch_free(batch); }
        } owned;
        auto &batch = owned.batch;
        require(batch.token && batch.pos && batch.n_seq_id && batch.seq_id && batch.logits,
                "batch allocation failed");
        for (int window = 0; window < 8; ++window) {
            llama_memory_clear(llama_get_memory(ctx.get()), true);
            const auto file = std::filesystem::path(argv[3]) / (std::to_string(window) + ".f32");
            const auto out = std::unique_ptr<FILE, decltype(&std::fclose)>(
                std::fopen(file.c_str(), "wb"), std::fclose);
            require(out != nullptr, "cannot open output");
            for (int start = 0; start < 1024; start += 128) {
                batch.n_tokens = 128;
                for (int i = 0; i < 128; ++i) {
                    require(batch.seq_id[i] != nullptr, "sequence allocation failed");
                    batch.token[i] = ids[static_cast<size_t>(window * 1024 + start + i)];
                    batch.pos[i] = start + i; batch.n_seq_id[i] = 1;
                    batch.seq_id[i][0] = 0; batch.logits[i] = 1;
                }
                require(llama_decode(ctx.get(), batch) == 0, "decode failed");
                for (int i = 0; i < 128; ++i) {
                    const float *logits = llama_get_logits_ith(ctx.get(), i);
                    require(logits != nullptr, "missing logits");
                    for (int j = 0; j < 128000; ++j)
                        require(std::isfinite(logits[j]), "nonfinite logits");
                    require(std::fwrite(logits, sizeof(float), 128000, out.get()) == 128000,
                            "incomplete logits write");
                }
            }
            require(std::fflush(out.get()) == 0, "output flush failed");
            std::fprintf(stderr, "RAW_WINDOW_COMPLETE %d 1024 128000\n", window);
        }
        std::puts("RAW_ALL_COMPLETE 8 1024 128000");
        return 0;
    } catch (const std::exception &error) {
        std::fprintf(stderr, "RAW_FAILED %s\n", error.what());
        return 1;
    }
}
