pub(crate) fn ar20_head_gates_eligible(
    enabled: bool,
    whole: bool,
    trace: bool,
    private_f32: bool,
    simd: u32,
    threads: u32,
) -> bool {
    enabled && whole && ar10_whole_gdn_eligible(true, trace, private_f32, simd, threads)
}

fn ar20_head_gates_require_all_actual_runtime_and_capability_guards() {
    let e: bool = kani::any();
    let w: bool = kani::any();
    let t: bool = kani::any();
    let f: bool = kani::any();
    let s: u32 = kani::any();
    let n: u32 = kani::any();
    let actual = ar20_head_gates_eligible(e, w, t, f, s, n);
    assert_eq!(actual, e && w && !t && f && s == 32 && n >= 512);
    kani::cover!(actual);
    kani::cover!(!actual && e && !w);
    kani::cover!(!actual && e && w && t);
}

pub(crate) fn ar10_whole_gdn_eligible(
    enabled: bool,
    trace: bool,
    private_fp32: bool,
    simd: u32,
    threads: u32,
) -> bool {
    gdn_private_epilogue_eligible(enabled, trace, private_fp32) && simd == 32 && threads >= 512
}

pub(crate) fn may_spin(enabled: bool, elapsed_ns: u128, polls: u32) -> bool {
    enabled && elapsed_ns < 50_000 && polls < 4096
}

fn gpu_spin_budget_never_exceeds_either_production_limit() {
    let enabled: bool = kani::any();
    let elapsed: u128 = kani::any();
    let polls: u32 = kani::any();
    let actual = may_spin(enabled, elapsed, polls);
    assert_eq!(actual, enabled && elapsed <= 49_999 && polls <= 4095);
    kani::cover!(actual && elapsed == 49_999 && polls == 4095);
    kani::cover!(!actual && enabled && elapsed == 50_000 && polls == 0);
    kani::cover!(!actual && enabled && elapsed == 0 && polls == 4096);
    kani::cover!(!actual && !enabled && elapsed == 0 && polls == 0);
}
