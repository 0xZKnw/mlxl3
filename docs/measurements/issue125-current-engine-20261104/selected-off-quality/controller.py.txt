"""Only the two selected-OFF final checks; qualified P13 process helpers are reused."""

import gzip
import hashlib
import json
import os
import runpy
import subprocess
import time
from pathlib import Path

ROOT = Path("/Users/justin/.codex/worktrees/issue125-current-engine/mix-stq1_0")
OUT = ROOT / "docs/measurements/issue125-current-engine-20261104/selected-off-quality"


def main():
    h = runpy.run_path(
        str(
            ROOT
            / "docs/measurements/issue120-hc-gdn-chain-20261104/final-quality/controller.py.txt"
        )
    )
    freeze = h["strict_json"](
        (ROOT / "docs/measurements/issue125-current-engine-20261104/freeze.json").read_text()
    )
    for item in freeze["artifacts"].values():
        assert hashlib.sha256(Path(item["path"]).read_bytes()).hexdigest() == item["sha256"]
    for file, sha in freeze["source_files"].items():
        assert hashlib.sha256((ROOT / file).read_bytes()).hexdigest() == sha, file
    assert (
        freeze["artifacts"]["direct-cli"]["sha256"]
        == "b4899081478369ae575dff4863d65570efb9909b384fdbbf6330a93a2e660110"
    )
    assert (
        freeze["artifacts"]["native-libtest"]["sha256"]
        == "7382bddbfb23538485bdb654e63319ab1513494189e67fd112e18f1e50df1296"
    )
    env = {k: v for k, v in os.environ.items() if not k.startswith(("MLXL3_", "DYLD_"))}
    env.update(
        {
            "DYLD_PRINT_LIBRARIES": "1",
            "PYTHONPATH": str(ROOT),
            "MLXL3_FLASH_MODEL": "/Users/justin/Documents/mix-stq1_0/models/Qwen3.8-Flash-Next-exl3-205-65c89531",
            "MLXL3_SSD_CACHE_GIB": "3",
            "MLXL3_SSD_READERS": "4",
            "MLXL3_SSD_DYNAMIC_READERS": "0",
            "MLXL3_SSD_LOOKAHEAD_DEPTH": "0",
            "MLXL3_SSD_OVERLAP": "1",
            "MLXL3_SSD_CACHE_POLICY": "auto",
            "MLXL3_SSD_PREDICTOR_DTYPE": "f32",
            "MLXL3_PROMPT_CACHE_MIB": "0",
        }
    )
    assert not any(k.startswith("MLXL3_NATIVE_") for k in env)
    OUT.mkdir(exist_ok=False)
    (OUT / "controller.py.txt").write_text(Path(__file__).read_text())
    lib = freeze["artifacts"]["native-libtest"]["path"]
    cli = freeze["artifacts"]["direct-cli"]["path"]
    phases = [
        (
            "native8",
            [
                lib,
                "gpu::flash::tests::native8_sampling_profile_preserves_outputs",
                "--exact",
                "--ignored",
                "--nocapture",
                "--test-threads=1",
            ],
            "I120_NATIVE8_PROFILE",
        ),
        (
            "cli",
            [
                "/Users/justin/Documents/mix-stq1_0/.venv/bin/python",
                "/Users/justin/.codex/worktrees/issue120-profile-next/mix-stq1_0/build/p01/essential/cli_short.py",
                cli,
                str(OUT),
            ],
            "NATIVE_SSD_DIRECT_CLI",
        ),
    ]
    report = {
        "status": "running",
        "parity": False,
        "all_children_joined": False,
        "artifacts": freeze["artifacts"],
        "environment": {k: v for k, v in env.items() if k.startswith(("MLXL3_", "PYTHONPATH"))},
        "phases": [],
    }

    def save():
        (OUT / "report.json").write_text(json.dumps(report, indent=2, allow_nan=False) + "\n")

    def journal(text):
        with (ROOT / "opti.md").open("a") as sink:
            sink.write("\nI125-SELECTED-DEFAULTS-OFF QUALITY " + text + "\n")

    save()
    try:
        for name, command, marker in phases:
            row = {
                "name": name,
                "command": command,
                "status": "running",
                "parity": False,
                "joined": False,
            }
            report["phases"].append(row)
            child = None
            start = time.monotonic()
            save()
            journal(name + " STARTED after explicit root grant")
            try:
                with (OUT / (name + ".log")).open("xb") as log:
                    child = subprocess.Popen(
                        command,
                        cwd=ROOT,
                        env=env,
                        stdout=log,
                        stderr=subprocess.STDOUT,
                        start_new_session=True,
                    )
                    row["pid"] = child.pid
                    save()
                    code = child.wait(timeout=180)
                assert code == 0 and not h["members"](child.pid)
                text = (OUT / (name + ".log")).read_text()
                a = h["marker"](text, marker)
                if name == "native8":
                    assert "test result: ok. 1 passed; 0 failed; 0 ignored;" in text
                    h["native_images"](text)
                    h["exact_fields"](
                        a,
                        {
                            "status": "diagnostic_complete",
                            "parity": True,
                            "prompt_tokens": 87,
                            "context": 4096,
                            "decode_rows": 7,
                            "cross_layer_triples": 238,
                            "hc_gdn_doubles": 14,
                            "gdn_separate": 0,
                            "profile_scope_ancestry_checked": True,
                            "independent_first_state": True,
                            "independent_first_logits": True,
                            "continuation_state_words": 28803072,
                        },
                    )
                    assert type(a["events"]) is list and a["events"]
                    assert (
                        type(a["tokens"]) is list
                        and len(a["tokens"]) == 8
                        and all(type(t) is int and 0 <= t < 248320 for t in a["tokens"])
                    )
                    for key in ["moe_wait_events", "moe_gpu_intervals"]:
                        assert type(a[key]) is int and a[key] > 0
                    requests = [
                        e["allocation_requests"]
                        for e in a["events"]
                        if e["name"].startswith("native.hc.workspace.")
                    ]
                    assert (
                        all(type(n) is int and n >= 0 for n in requests) and sum(requests) == 2716
                    )
                    (OUT / "native8-profile.json.gz").write_bytes(
                        gzip.compress(
                            json.dumps(a, allow_nan=False, separators=(",", ":")).encode(),
                            compresslevel=1,
                            mtime=0,
                        )
                    )
                    a = {k: v for k, v in a.items() if k != "events"} | {
                        "events": len(a["events"]),
                        "workspace_allocation_requests": 2716,
                        "status": "validated_selected_defaults_diagnostic_only",
                    }
                    (OUT / "native8-safe-summary.json").write_text(
                        json.dumps(a, indent=2, allow_nan=False) + "\n"
                    )
                else:
                    h["exact_fields"](
                        a,
                        {
                            "status": "verified",
                            "parity": True,
                            "completed": 1,
                            "context_length": 4096,
                            "token_hash": "9f850af1cbba074b",
                            "mlx_loaded": False,
                            "child_joined": True,
                        },
                    )
                    ledger = h["strict_json"]((OUT / "cli4096-short/report.json").read_text())[
                        "ledger"
                    ]
                    assert "hc_workspace" not in ledger["state_backing_bytes"]
                    slots = ledger["expert_slots_by_layer"]
                    assert (
                        type(slots) is list
                        and len(slots) == 48
                        and all(type(n) is int and n == 53 for n in slots)
                    )
                row.update(
                    status="validated",
                    parity=True,
                    joined=True,
                    seconds=time.monotonic() - start,
                    evidence=a,
                )
                save()
                journal(name + " VALIDATED")
                print(name + " validated", flush=True)
            except BaseException as error:
                row.update(
                    status="failed", parity=False, joined=h["stop"](child), error=repr(error)
                )
                save()
                raise
        assert len(report["phases"]) == 2 and all(
            p["parity"] is True and p["joined"] is True for p in report["phases"]
        )
        report.update(status="validated", parity=True, all_children_joined=True)
        journal("2/2 VALIDATED selected defaults, no speed measurement")
    except BaseException as error:
        report.update(
            status="failed",
            parity=False,
            error=repr(error),
            all_children_joined=all(p["joined"] for p in report["phases"]),
        )
        journal("FAILED " + repr(error))
        raise
    finally:
        save()


if __name__ == "__main__":
    main()
