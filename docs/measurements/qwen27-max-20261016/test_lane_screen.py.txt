def test_candidate_wrapper_permutates_only_candidates_and_reuses_metadata(monkeypatch):
    class CpuArray:
        int32 = np.int32

        def __init__(self):
            self.allocations = 0
            self.takes = 0

        def array(self, values, dtype):
            self.allocations += 1
            return np.asarray(values, dtype=dtype)

        def take(self, value, indices, axis):
            self.takes += 1
            assert axis == -1
            return np.take(value, indices, axis=axis)

    def factory(mx, shape, rows, **options):
        return lambda xhat, weights, tile_sub: xhat

    monkeypatch.setattr(screen, "BASE_QMV", factory)
    for rows in (1, 2):
        for groups in (1, 2, 3):
            mx = CpuArray()
            shape = {"input": 128, "widths": [128] * groups}
            original = np.arange(rows * groups * 128, dtype=np.int32).reshape(rows, groups, 128)
            candidate = screen.qmv(mx, shape, rows, source=screen.MARKER)
            stock = screen.qmv(mx, shape, rows, source="stock")
            assert mx.allocations == 1 and mx.takes == 0
            for _ in range(2):
                actual = candidate(original, None, None)
                for row in range(rows):
                    for group in range(groups):
                        for tile in range(8):
                            for lane in range(32):
                                begin = tile * 16
                                expected = [
                                    original[row, group, begin + 2 * (lane % 4) + offset]
                                    for offset in (0, 1, 8, 9)
                                ]
                                np.testing.assert_array_equal(
                                    actual[
                                        row,
                                        group,
                                        begin + 4 * (lane % 4) : begin + 4 * (lane % 4) + 4,
                                    ],
                                    expected,
                                )
                np.testing.assert_array_equal(stock(original, None, None), original)
            assert mx.allocations == 1 and mx.takes == 2
    with pytest.raises(ValueError, match="ambiguous"):
        screen.qmv(mx, shape, rows, source=screen.MARKER * 2)

@pytest.mark.parametrize(
    "error", [ValueError("divergent"), TimeoutError("silent"), KeyboardInterrupt()]
)
def test_real_cli_preserves_failures_and_restores_base(tmp_path, monkeypatch, error):
    output = tmp_path / "failed.json"
    originals = (
        screen.base.run,
        screen.base.self_test,
        screen.base.transform,
        screen.base.qmv,
        screen.base.stock_nt,
    )

    def fail(args, report, save):
        assert screen.base.transform is screen.transform and screen.base.qmv is screen.qmv
        report["checks"].append({"fixture": "observed"})
        report.update(status="complete", parity=True)
        save()
        raise error

    monkeypatch.setattr(screen, "BASE_RUN", fail)
    monkeypatch.setattr(
        sys, "argv", ["lane", "--checkpoint", str(tmp_path), "--output", str(output)]
    )
    with pytest.raises(type(error)):
        screen.main()
    report = json.loads(output.read_text())
    assert report["status"] == "failed" and report["parity"] is False
    assert report["checks"] == [{"fixture": "observed"}] and report["error"]
    assert report["candidate"] == "half4_lane_order"
    assert (
        screen.base.run,
        screen.base.self_test,
        screen.base.transform,
        screen.base.qmv,
        screen.base.stock_nt,
    ) == originals


def test_real_cli_success_and_source_wiring(tmp_path, monkeypatch):
    output = tmp_path / "complete.json"

    def complete(args, report, save):
        for name in ("_qmv_tile_kernel.metal", "_qmv_mapped_tile_kernel.metal"):
            source = (screen.base.ROOT / "native/shaders" / name).read_text()
            candidate = screen.base.transform(source)
            assert screen.MARKER in candidate and screen.base.NEW not in candidate
        assert screen.base.qmv is screen.qmv
        assert screen.base.stock_nt is screen.stock_nt
        report.update(status="complete", parity=True)

    monkeypatch.setattr(screen, "BASE_RUN", complete)
    monkeypatch.setattr(
        sys, "argv", ["lane", "--checkpoint", str(tmp_path), "--output", str(output)]
    )
    screen.main()
    report = json.loads(output.read_text())
    assert report["status"] == "complete" and report["parity"] is True
    assert report["lane_order"] == list(screen.LANE_ORDER)


def test_existing_evidence_cannot_be_overwritten(tmp_path, monkeypatch):
    output = tmp_path / "existing.json"
    output.write_bytes(b"preserve")
    monkeypatch.setattr(
        sys, "argv", ["lane", "--checkpoint", str(tmp_path), "--output", str(output)]
    )
    with pytest.raises(SystemExit, match="2"):
        screen.main()
    assert output.read_bytes() == b"preserve"
