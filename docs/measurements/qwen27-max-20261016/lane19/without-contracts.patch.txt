+fn lane_order_indices(
+    input: i32,
+    output: i32,
+    k: usize,
+    cb: Codebook,
+    grouped: bool,
+) -> Result<Option<Array>> {
+    if !crate::contracts::exl3_lane_order(
+        1, input, output, k, grouped, cb == Codebook::Mul1,
+        array::is_m5_gpu()?, lane_order_enabled(),
+    ) {
+        return Ok(None);
+    }
+    let values = (0..input)
+        .map(|index| i32::try_from(crate::contracts::exl3_lane_source_index(index as u32)))
+        .collect::<std::result::Result<Vec<_>, _>>()?;
+    let indices = Array::from_i32(&values, &[input])?;
+    indices.eval()?;
+    Ok(Some(indices))
+}

+        Ok((xhat, false))
+    }
