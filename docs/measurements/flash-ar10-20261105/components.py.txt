def validate(helpers, text, phase, fixture_count):
    assert "running 1 test" in text
    assert "test result: ok. 1 passed; 0 failed; 0 ignored;" in text
    helpers["native_images"](text)
    evidence = helpers["marker"](text, phase[2])
    helpers["exact_fields"](evidence, {"parity": True, **phase[4]})
    if phase[0] == "A01":
        assert type(evidence["compared_bytes"]) is int and evidence["compared_bytes"] > 0
    if phase[0] == "A02":
        helpers["exact_fields"](evidence, {"finite_stock_half_words": (fixture_count - 1) * 4 * 2 * 2560})
        values = evidence["samples_seconds_per_expert"]
        assert len(values) == 4 and all(len(row) == 6 for row in values)
        assert all(type(v) in (int, float) and v > 0 for row in values for v in row)
        medians = evidence["seconds_per_expert_medians"]
        assert len(medians) == 4 and all(type(v) in (int, float) and v > 0 for v in medians)
    if phase[0] == "A03-05":
        assert type(evidence["up_finish_uses"]) is int and evidence["up_finish_uses"] > 0
    return evidence

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--freeze", type=Path, default=ROOT / "build/ar10/freeze.json")
    parser.add_argument("--out", type=Path, default=ROOT / "docs/measurements/flash-ar10-20261105/components-01")
    args = parser.parse_args()
    args.out.mkdir(parents=True, exist_ok=True)
    sink = args.out / "report.json"
    assert not sink.exists(), "refusing to overwrite prior evidence"
    report = {"status": "preflight", "parity": False, "all_children_joined": True, "phases": []}
    def save():
        temporary = sink.with_suffix(".tmp")
        temporary.write_text(json.dumps(report, indent=2, allow_nan=False) + "\n")
        temporary.replace(sink)
    def journal(message):
        with (ROOT / "opti.md").open("a") as stream:
            stream.write("\nAR10 COMPONENTS " + message + "\n")
    save()
    try:
        helpers = runpy.run_path(str(OLD))
        freeze = helpers["strict_json"](args.freeze.read_text())
        lib = ROOT / "build/ar10/frozen/native-libtest"
        artifact = freeze["artifacts"]["native-libtest"]
        assert Path(artifact["path"]).resolve() == lib.resolve()
        assert hashlib.sha256(lib.read_bytes()).hexdigest() == artifact["sha256"]
        for name, expected in freeze["source_files"].items():
            assert hashlib.sha256((ROOT / name).read_bytes()).hexdigest() == expected, name
        cases = helpers["strict_json"]((FIXTURES / "cases.json").read_text())
        assert type(cases) is list and len(cases) >= 5
        report.update(helper_sha256=hashlib.sha256(OLD.read_bytes()).hexdigest(), artifact=artifact, fixture_count=len(cases), status="running")
        env = {k: v for k, v in os.environ.items() if not k.startswith(("MLXL3_", "DYLD_"))}
        env.update({"DYLD_PRINT_LIBRARIES": "1", "PYTHONPATH": str(ROOT), "MLXL3_FLASH_MODEL": MODEL, "MLXL3_ROUTED_SCRATCH_FIXTURES": str(FIXTURES), "MLXL3_SSD_CACHE_GIB": "3", "MLXL3_SSD_READERS": "4", "MLXL3_SSD_LOOKAHEAD_DEPTH": "0", "MLXL3_SSD_OVERLAP": "1", "MLXL3_SSD_CACHE_POLICY": "auto", "MLXL3_PROMPT_CACHE_MIB": "0", "MLXL3_SSD_DYNAMIC_READERS": "0"})
        for key in ["HC_WORKSPACE", "HC_UP_MIX", "HC_DOWN_INJECT", "SSD_EXTRA_WAVE", "GDN_HALF_ELISION", "GDN_NORM_GATE", "FIXED_RESIDENCY", *NEW]:
            env["MLXL3_NATIVE_" + key] = "0"
        for phase in PHASES:
            runenv = dict(env)
            for key in phase[3]:
                runenv["MLXL3_NATIVE_" + key] = "1"
            command = [str(lib), phase[1], "--exact", "--ignored", "--nocapture", "--test-threads=1"]
            item = {"label": phase[0], "status": "running", "parity": False, "joined": False, "command": command, "flags": {k: v for k, v in runenv.items() if k.startswith("MLXL3_NATIVE_")}}
            report["phases"].append(item)
            report["all_children_joined"] = False
            save()
            journal(phase[0] + " STARTED")
            child, failure = None, None
            started = time.monotonic()
            try:
                path = args.out / (phase[0] + ".log")
                with path.open("x") as log:
                    child = subprocess.Popen(command, cwd=ROOT, env=runenv, stdout=log, stderr=subprocess.STDOUT, start_new_session=True)
                    item["pid"] = child.pid
                    save()
                    code = child.wait(timeout=180)
                assert code == 0 and not helpers["members"](child.pid), code
                item["evidence"] = validate(helpers, path.read_text(), phase, len(cases))
            except Exception as error:
                failure = repr(error)
            finally:
                item["joined"] = helpers["stop"](child)
                item["seconds"] = time.monotonic() - started
                report["all_children_joined"] = all(p["joined"] for p in report["phases"])
                save()
            assert item["joined"], "child cleanup failed; refusing later phases"
            if failure is not None:
                item.update(status="failed", parity=False, error=failure)
                save()
                journal(phase[0] + " FAILED " + failure)
                continue
            item.update(status="validated", parity=True)
            save()
            journal(phase[0] + " VALIDATED")
        assert len(report["phases"]) == len(PHASES) and report["all_children_joined"]
        passed = sum(p["status"] == "validated" for p in report["phases"])
        report.update(status="validated" if passed == len(PHASES) else "completed_with_failures", parity=passed == len(PHASES))
        journal(f"{passed}/10 VALIDATED components only; no model-speed claim")
    except BaseException as error:
        report.update(status="failed", parity=False, error=repr(error))
        if report["phases"] and report["phases"][-1]["status"] == "running":
            report["phases"][-1]["status"] = "failed"
        journal("FAILED " + repr(error))
        raise
    finally:
        save()

if __name__ == "__main__":
    main()
