class BoundedReader:
    def __init__(self):
        self.remaining = 3 * (1 << 20) + 17

    def open(self, mode):
        assert mode == "rb"
        return self

    def __enter__(self):
        return self

    def __exit__(self, *args):
        return False

    def read(self, size):
        assert 0 < size <= 1 << 20
        count = min(size, self.remaining)
        self.remaining -= count
        return b"a" * count
