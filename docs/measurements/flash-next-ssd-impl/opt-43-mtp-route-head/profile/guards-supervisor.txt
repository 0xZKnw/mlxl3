cases = []


def check(prev, profiles, accepted, case):
    log = "\n".join("SSD_LIGHT_PROFILE " + json.dumps(v) for v in profiles)
    result = {}
    try:
        checked(log, prev, records, prev["name"], result)
        observed = True
    except (AssertionError, ValueError, TypeError, KeyError, StopIteration, OverflowError):
        observed = False
    assert observed is accepted, case
    assert observed == ("diagnostics" in result), "partial validation leaked"
    cases.append({"case": prev["name"] + ":" + case, "accepted": observed})


for prev in records:
    if not prev["name"].startswith("light-") or prev["status"] != "passed":
        continue
    profiles = prev["diagnostics"]
    check(prev, profiles, True, "valid")
    check(prev, [], False, "empty")
    check(prev, profiles[:1], False, "incomplete")
    mutations = [
        (["phase"], "wrong"),
        (["tokens"], []),
        (["tokens", 0], True),
        (["tokens", 0], 248320),
        (["text"], ""),
        (["text"], "wrong"),
        (["logit_count"], 0),
        (["logits_finite"], False),
        (["route_fill_import_seconds", 0], float("nan")),
        (["candidate_scanned_hot_admitted_imported_overlap_rows", 0], -1),
        (["memory", "process_lifetime_peak_bytes"], 0),
        (["mtp"], not profiles[0]["mtp"]),
        (["mtp_proposed_tokens"], 999999),
    ]
    if profiles[0]["lookahead_depth"]:
        mutations += [
            (["prefetch", "completed"], -1),
            (["memory", "process_lifetime_peak_bytes"], 1 << 60),
        ]
    else:
        mutations += [
            (["prefetch"], {}),
            (["candidate_scanned_hot_admitted_imported_overlap_rows", 3], 1),
        ]
    for keys, value in mutations:
        altered = copy.deepcopy(profiles)
        target = altered[0]
        for key in keys[:-1]:
            target = target[key]
        target[keys[-1]] = value
        check(prev, altered, False, str(keys))

def vm(page=16384, count=0):
    return f"Mach Virtual Memory Statistics: (page size of {page} bytes)\n" + "".join(
        f"{key}: {count}.\n"
        for key in [
            "Pages free",
            "File-backed pages",
            "Pages occupied by compressor",
            "Pageins",
            "Compressions",
            "Decompressions",
            "Swapins",
            "Swapouts",
        ]
    )


for page in [4096, 16384, 65536]:
    assert set(delta(vm(page, 2), vm(page, 5)).values()) == {3 * page}
    assert set(delta(vm(page, 2), vm(page, 2)).values()) == {0}
for before, after in [
    (vm(), vm(0)),
    (vm(), vm(4096)),
    (vm(count=2), vm(count=1)),
    (vm(), ""),
    (vm(), vm() + "Pageins: 0.\n"),
    (vm(), vm().replace("Pageins: 0.\n", "")),
]:
    try:
        delta(before, after)
    except ValueError:
        pass
    else:
        raise AssertionError("invalid VM events accepted")
