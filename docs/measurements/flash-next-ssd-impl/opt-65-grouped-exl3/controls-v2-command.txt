import json, runpy, subprocess, time, os
from pathlib import Path
base=Path('docs/measurements/flash-next-ssd-impl/opt-65-grouped-exl3')
phase=os.environ['FLASH_CONTROL_PHASE']; assert phase in ('before','after')
p=base/('control-v2-'+phase)
plan=p/'run.json'
api=runpy.run_path('scripts/profile_flash.py')
parse=runpy.run_path('benchmarks/benchmark_flash_memory.py')['parse_vm']
def take():
 return {'epoch':time.time(),'vm_bytes':parse(subprocess.check_output(['vm_stat'],text=True,timeout=5)), 'power':subprocess.check_output(['pmset','-g','batt'],text=True,timeout=5)}
start=take(); assert start['vm_bytes']['Swapouts']==121528483840; baseline={k:start['vm_bytes'][k] for k in ('Swapins','Swapouts')}
origin=p/'origin.json'; assert not origin.exists(); origin.write_text(json.dumps(start,indent=2)+'\n')
result={'status':'incomplete','parity':False,'original_vm_baseline':baseline,'source_identity':'identity-v2.json','source_commit':'5443cd05 + selective EXL3 v2'}
identity=json.loads((base/'identity-v2.json').read_text())
import hashlib
for f in identity['files']+identity['artifacts']: assert hashlib.sha256(Path(f['path']).read_bytes()).hexdigest()==f['sha256'],f['path']
def watch(seconds,path):
 records=[]
 try:
  deadline=time.monotonic()+seconds
  while True:
   row=take(); records.append(row)
   assert row['power'].splitlines()[0]==start['power'].splitlines()[0]
   assert row['power'].splitlines()[0]=="Now drawing from 'AC Power'"
   for key in baseline:
    assert 0<=row['vm_bytes'][key]-baseline[key]<=256<<20
   assert row['vm_bytes']['Swapouts']==baseline['Swapouts'], 'new Swapout'
   if time.monotonic()>=deadline: break
   time.sleep(min(2,deadline-time.monotonic()))
 finally:
  path.write_text(json.dumps({'samples':records},indent=2)+'\n')
try:
 assert all(json.loads((base/phase/'campaign-result.json').read_text())['status']=='passed_quality_and_postflight30s' for phase in ['quality-slots-v2','quality-target-v2','quality-mtp-v2','micro-v2'])
 assert json.loads((base/'micro-v2/qualification.json').read_text())['full_model_eligible'] is True
 if phase=='after': assert json.loads((base/'speed-result-v2.json').read_text())['status']=='completed_speed_and_postflight30s'
 watch(20,p/'preflight.json')
 print('65T preflight20s passed; controls start',flush=True)
 assert all(item['env']['MLXL3_FLASH_MULTI_TOKEN']=='1' for item in json.loads(plan.read_text()))
 api['run_sampler'](plan,baseline)
 raw=json.loads(plan.with_name('run-result.json').read_text())
 assert raw['status']=='completed_expected_outcomes_not_speed' and len(raw['commands'])==4
 assert all(r['reaped'] and r['status']=='completed_expected_outcome' and r['swapouts_bytes']==0 for r in raw['commands'])
 for item in json.loads(plan.read_text()):
  report=json.loads((p/item['name']/'report.json').read_text()); assert report['status']=='validated' and report['reaped'] and report['reader_joined'] and len(report['runs'])==len(report['warmup'])==1
  golden=json.loads(Path('docs/measurements/flash-next-ssd-impl/opt-60-splash-dflash/speed-v7-final',item['prompt']+'-0-OFF','report.json').read_text())['runs'][0]
  for run in report['runs']+report['warmup']:
   assert (run['token_hash'],run['text'])==(golden['token_hash'],golden['text'])
   assert run['stats']['decode_tokens']==127 and 0<run['stats']['memory']['process_lifetime_peak_bytes']<=12_000_000_000
 result.update(status='passed_control_pending_postflight',parity=True)
except BaseException as e:
 result.update(status='failed_or_interrupted',parity=False,error=repr(e)); raise
finally:
 try:
  watch(30,p/'postflight.json')
  if result['parity']: result['status']='passed_control_and_postflight30s'
 except BaseException as e:
  result.update(status='failed_postflight',parity=False,postflight_error=repr(e)); raise
 finally:
  (p/'campaign-result.json').write_text(json.dumps(result,indent=2)+'\n')
  print(result,flush=True)
