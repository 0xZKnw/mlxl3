from pathlib import Path
import os, subprocess, json, time, signal, re, runpy, hashlib, math

w = Path.cwd()
root = Path("/Users/justin/Documents/mix-stq1_0")
p = w / "docs/measurements/flash-next-ssd-impl/opt-34-les-ready"
old = json.loads(
    (
        w / "docs/measurements/flash-next-ssd-impl/opt-27-shared-overlap/repair/quality.json"
    ).read_text()
)
ident = json.loads((p / "identity.json").read_text())
parse_vm = runpy.run_path(str(w / "benchmarks/benchmark_flash_memory.py"))["parse_vm"]
records = []

assert json.loads((p.parent / "opt-30-les/download.json").read_text())["status"] == "verified", (
    "model restoration not verified"
)
for previous in old:
    previous["env"]["MLXL3_SSD_LOOKAHEAD_DEPTH"] = "4"
    if previous["name"] == "ssd":
        previous["expected_tests"] = 8
extra = {
    **old[1],
    "name": "les-full-state",
    "command": [
        "/Users/justin/Documents/mix-stq1_0/build/flash-opt-34/les-ready/mlxl3_native-28585053ede733ec",
        "qwen4::tests::flash_lookahead_preserves_full_logits_states_and_pair_commits",
        "--exact",
        "--ignored",
        "--test-threads=1",
        "--nocapture",
    ],
    "expected_tests": 1,
    "timeout_seconds": 240,
}
old.insert(0, extra)
copy_check = {
    **extra,
    "name": "staged-ffi",
    "command": [
        extra["command"][0],
        "array::tests::staged_records_copy_exact_disjoint_bytes_and_reject_invalid_ranges",
        "--exact",
        "--ignored",
        "--test-threads=1",
        "--nocapture",
    ],
}
old.insert(0, copy_check)
for mtp in [False, True]:
    for prompt in ["MAIN", "HOLD"]:
        for depth in [0, 4]:
            profile = {
                **extra,
                "name": f"light-{'mtp' if mtp else 'baseline'}-{prompt}-{depth}",
                "command": [
                    extra["command"][0],
                    "qwen4::tests::flash_light_profile_preserves_natural_decode",
                    "--exact",
                    "--ignored",
                    "--test-threads=1",
                    "--nocapture",
                ],
                "env": {
                    **extra["env"],
                    "MLXL3_SSD_LOOKAHEAD_DEPTH": str(depth),
                    "MLXL3_FLASH_PROFILE_MTP": str(int(mtp)),
                    "MLXL3_FLASH_PROFILE_PROMPT": prompt,
                },
                "timeout_seconds": 360,
            }
            old.insert(-1, profile)
assert len(old) == 24
for path, h in ident["binaries"].items():
    assert hashlib.sha256(Path(path).read_bytes()).hexdigest() == h
for path, h in ident["files"].items():
    assert hashlib.sha256((w / path).read_bytes()).hexdigest() == h


def snapshot():
    return {
        key: subprocess.check_output(cmd, text=True, timeout=5)
        for key, cmd in [
            ("power", ["pmset", "-g", "batt"]),
            ("vm_stat", ["vm_stat"]),
            ("swap", ["sysctl", "vm.swapusage"]),
        ]
    }


try:
    for prev in old:
        name = prev["name"]
        cmd = [
            x.replace("flash-opt-27/shared-overlap-repaired", "flash-opt-34/les-ready")
            for x in prev["command"]
        ]
        limit = prev["timeout_seconds"]
        r = {
            "name": name,
            "command": cmd,
            "env": prev["env"],
            "timeout_seconds": limit,
            "expected_tests": prev["expected_tests"],
            "status": "running",
            "before": snapshot(),
        }
        assert r["before"]["power"].splitlines()[0] == "Now drawing from 'AC Power'", (
            "quality requires AC"
        )
        records.append(r)
        (p / "quality.json").write_text(json.dumps(records, indent=2) + "\n")
        child = None
        t = time.monotonic()
        try:
            with (p / (name + ".log")).open("w") as f:
                child = subprocess.Popen(
                    cmd,
                    env={**os.environ, **prev["env"]},
                    stdout=f,
                    stderr=subprocess.STDOUT,
                    start_new_session=True,
                )
                r["pid"] = child.pid
                r["exit_code"] = child.wait(timeout=limit)
            assert r["exit_code"] == 0, "quality process failed"
            log = (p / (name + ".log")).read_text()
            if r["expected_tests"] is not None:
                counts = re.findall(r"test result: ok[.] (\d+) passed; 0 failed;", log)
                assert counts and sum(map(int, counts)) == r["expected_tests"], (
                    "absent or incorrect test count"
                )
                r["observed_tests"] = sum(map(int, counts))
                if name.startswith("light-"):
                    profiles = [
                        json.loads(line.split("SSD_LIGHT_PROFILE ", 1)[1])
                        for line in log.splitlines()
                        if "SSD_LIGHT_PROFILE " in line
                    ]
                    assert len(profiles) == 2 and [v["phase"] for v in profiles] == [
                        "warmup",
                        "measured",
                    ]
                    mtp = prev["env"]["MLXL3_FLASH_PROFILE_MTP"] == "1"
                    prompt = prev["env"]["MLXL3_FLASH_PROFILE_PROMPT"]
                    depth = int(prev["env"]["MLXL3_SSD_LOOKAHEAD_DEPTH"])
                    oracle = json.loads(
                        (
                            p.parent / f"opt-18-hc/speed/1-{prompt}-baseline-H/report.json"
                        ).read_text()
                    )["runs"][0]
                    for v in profiles:
                        assert (
                            v["status"] == "completed_diagnostic_awaiting_external_oracle"
                            and v["mtp"] is mtp
                            and v["prompt"] == prompt
                            and v["lookahead_depth"] == depth
                        )
                        ids = v["tokens"]
                        assert len(ids) == 64 and all(
                            type(i) is int and 0 <= i < 248320 for i in ids
                        )
                        h = 0xCBF29CE484222325
                        for token in ids:
                            for byte in token.to_bytes(4, "little"):
                                h = ((h ^ byte) * 0x100000001B3) & ((1 << 64) - 1)
                        assert (
                            f"{h:016x}" == oracle["token_hash"] and v["text"] == oracle["text"]
                        ), "diagnostic H20 hash/text diverged"
                        assert v["logit_count"] == 248320 and v["logits_finite"] is True
                        assert len(v["route_fill_import_seconds"]) == 3
                        assert all(
                            type(x) in [int, float] and math.isfinite(x) and x >= 0
                            for x in v["route_fill_import_seconds"]
                            + v["mtp_head_verify_repair_total_seconds"]
                        )
                        counts = v["candidate_scanned_hot_admitted_imported_overlap_rows"]
                        assert len(counts) == 6 and all(type(x) is int and x >= 0 for x in counts)
                        peak = v["memory"]["process_lifetime_peak_bytes"]
                        assert type(peak) is int and peak > 0
                        if depth:
                            assert (
                                v["prefetch"]["submitted"] > 0
                                and v["prefetch"]["completed"] == v["prefetch"]["submitted"]
                                and counts[3] > 0
                            )
                            control = next(q for q in records if q["name"] == name[:-1] + "0")
                            control_peak = next(
                                q for q in control["diagnostics"] if q["phase"] == v["phase"]
                            )["memory"]["process_lifetime_peak_bytes"]
                            assert peak <= control_peak + (256 << 20), (
                                "diagnostic extra process peak exceeds limit"
                            )
                        else:
                            assert v["prefetch"] is None and counts[:5] == [0] * 5
                        if mtp:
                            expected = json.loads(
                                (
                                    p.parent / f"opt-32-les-async/mtp-screen/{prompt}-counters.json"
                                ).read_text()
                            )
                            assert all(
                                v[k] == expected[k]
                                for k in [
                                    "mtp_proposed_tokens",
                                    "mtp_accepted_tokens",
                                    "mtp_blocks",
                                ]
                            )
                        else:
                            assert all(
                                v[k] == 0
                                for k in [
                                    "mtp_proposed_tokens",
                                    "mtp_accepted_tokens",
                                    "mtp_blocks",
                                ]
                            )
                    r["diagnostics"] = profiles
            else:
                summary = json.loads(log.strip().splitlines()[-1])
                assert summary == {
                    "status": "validated",
                    "budgets": [1, 2, 3, 17],
                    "cancel_checked": True,
                }
                r["summary"] = summary
            r["after"] = snapshot()
            assert r["after"]["power"].splitlines()[0] == r["before"]["power"].splitlines()[0], (
                "power changed"
            )
            r["swapout_bytes"] = (
                parse_vm(r["after"]["vm_stat"])["Swapouts"]
                - parse_vm(r["before"]["vm_stat"])["Swapouts"]
            )
            assert 0 <= r["swapout_bytes"] <= 256 << 20, "quality swap limit exceeded"
            r["status"] = "passed"
        except BaseException as exc:
            r.update(
                status="interrupted" if isinstance(exc, KeyboardInterrupt) else "failed",
                error=repr(exc),
            )
            raise
        finally:
            if child is not None and child.poll() is None:
                os.killpg(child.pid, signal.SIGTERM)
                try:
                    child.wait(timeout=5)
                except subprocess.TimeoutExpired:
                    os.killpg(child.pid, signal.SIGKILL)
                    child.wait()
            r["reaped"] = child is None or child.poll() is not None
            r["seconds"] = time.monotonic() - t
            (p / "quality.json").write_text(json.dumps(records, indent=2) + "\n")
            with (w / "opti.md").open("a") as f:
                f.write(
                    "\n- 34 qualité "
                    + name
                    + " : **"
                    + r["status"]
                    + "**, swapouts+"
                    + str(r.get("swapout_bytes", "non mesuré"))
                    + "octets ; quality.json/log.\n"
                )
            print(
                {k: r[k] for k in ["name", "status", "seconds", "swapout_bytes"] if k in r},
                flush=True,
            )
finally:
    (p / "quality.json").write_text(json.dumps(records, indent=2) + "\n")
