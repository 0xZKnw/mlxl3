//! Bounded CPU staging; the inference thread never waits for a prediction.
use crate::{checkpoint::TensorInfo, flash_support::EXPERT_BYTES};
use anyhow::{Result, ensure};
use std::{
    os::unix::fs::FileExt,
    sync::{
        Arc,
        atomic::{AtomicU64, Ordering},
        mpsc::{self, Receiver, SyncSender, TrySendError},
    },
    thread::{self, JoinHandle},
};

#[derive(Default)]
pub(crate) struct Counters {
    pub submitted: AtomicU64,
    pub completed: AtomicU64,
    pub used: AtomicU64,
    pub discarded: AtomicU64,
    pub failed: AtomicU64,
    pub late_needed: AtomicU64,
    pub bytes: AtomicU64,
}

struct Job {
    layer: usize,
    expert: u32,
    source: TensorInfo,
    bytes: Vec<u8>,
}
struct Completed {
    job: Job,
    valid: bool,
}
struct Worker {
    tx: Option<SyncSender<Job>>,
    rx: Receiver<Completed>,
    thread: Option<JoinHandle<()>>,
    pending: Option<(usize, u32)>,
    buffer: Option<Vec<u8>>,
    #[cfg(test)]
    ready: Option<Completed>,
}
pub(crate) struct Staged {
    pub expert: u32,
    pub bytes: Vec<u8>,
    worker: usize,
}
pub(crate) struct ReaderPool {
    workers: Vec<Worker>,
    pub counters: Arc<Counters>,
}

fn read(job: &mut Job, counters: &Counters) -> Result<()> {
    ensure!(
        job.bytes.len() == EXPERT_BYTES as usize,
        "invalid staging buffer"
    );
    let (file, offset) = job.source.source()?;
    ensure!(
        offset.checked_add(EXPERT_BYTES).is_some_and(
            |end| end <= i64::MAX as u64 && file.metadata().is_ok_and(|m| end <= m.len())
        ),
        "invalid speculative record range"
    );
    let mut done = 0;
    while done < job.bytes.len() {
        match file.read_at(&mut job.bytes[done..], offset + done as u64) {
            Ok(0) => anyhow::bail!("speculative checkpoint truncated"),
            Ok(n) => {
                done += n;
                counters.bytes.fetch_add(n as u64, Ordering::Relaxed);
            }
            Err(e) if e.kind() == std::io::ErrorKind::Interrupted => continue,
            Err(e) => return Err(e.into()),
        }
    }
    Ok(())
}

impl ReaderPool {
    pub fn new() -> Result<Self> {
        let mut pool = Self {
            workers: Vec::new(),
            counters: Arc::default(),
        };
        for index in 0..4 {
            let (tx, jobs) = mpsc::sync_channel::<Job>(1);
            let (results, rx) = mpsc::sync_channel(1);
            let counters = Arc::clone(&pool.counters);
            let thread = thread::Builder::new()
                .name(format!("ssd-lookahead-{index}"))
                .spawn(move || {
                    while let Ok(mut job) = jobs.recv() {
                        let valid = read(&mut job, &counters).is_ok();
                        counters.completed.fetch_add(1, Ordering::Relaxed);
                        if !valid {
                            counters.failed.fetch_add(1, Ordering::Relaxed);
                        }
                        if results.send(Completed { job, valid }).is_err() {
                            break;
                        }
                    }
                })?;
            pool.workers.push(Worker {
                tx: Some(tx),
                rx,
                thread: Some(thread),
                pending: None,
                buffer: Some(vec![0; EXPERT_BYTES as usize]),
                #[cfg(test)]
                ready: None,
            });
        }
        Ok(pool)
    }

    pub fn submit(&mut self, layer: usize, expert: u32, source: TensorInfo) {
        if expert >= 512
            || self
                .workers
                .iter()
                .any(|w| w.pending == Some((layer, expert)))
        {
            return;
        }
        let Some(worker) = self
            .workers
            .iter_mut()
            .find(|w| w.pending.is_none() && w.buffer.is_some() && w.tx.is_some())
        else {
            return;
        };
        let bytes = worker.buffer.take().expect("available staging buffer");
        match worker.tx.as_ref().expect("available worker").try_send(Job {
            layer,
            expert,
            source,
            bytes,
        }) {
            Ok(()) => {
                worker.pending = Some((layer, expert));
                self.counters.submitted.fetch_add(1, Ordering::Relaxed);
            }
            Err(TrySendError::Full(job)) => worker.buffer = Some(job.bytes),
            Err(TrySendError::Disconnected(job)) => {
                worker.buffer = Some(job.bytes);
                worker.tx = None;
            }
        }
    }

    pub fn take(&mut self, layer: usize, ids: &[u32]) -> Vec<Staged> {
        let mut staged = Vec::new();
        for (index, worker) in self.workers.iter_mut().enumerate() {
            #[cfg(test)]
            let completed = worker.ready.take().or_else(|| worker.rx.recv().ok());
            #[cfg(not(test))]
            let completed = worker.rx.recv().ok();
            let Some(completed) = completed else {
                if worker.pending.is_some_and(|(pending_layer, expert)| {
                    pending_layer == layer && ids.contains(&expert)
                }) {
                    self.counters.late_needed.fetch_add(1, Ordering::Relaxed);
                }
                continue;
            };
            worker.pending = None;
            let job = completed.job;
            if completed.valid && job.layer == layer && ids.contains(&job.expert) {
                staged.push(Staged {
                    expert: job.expert,
                    bytes: job.bytes,
                    worker: index,
                });
            } else {
                self.counters.discarded.fetch_add(1, Ordering::Relaxed);
                worker.buffer = Some(job.bytes);
            }
        }
        staged
    }

    #[cfg(test)]
    pub fn wait_for_test(&mut self) -> Result<()> {
        for worker in &mut self.workers {
            if worker.pending.is_some() && worker.ready.is_none() {
                worker.ready = Some(worker.rx.recv_timeout(std::time::Duration::from_secs(5))?);
            }
        }
        Ok(())
    }

    pub fn recycle(&mut self, staged: Staged, used: bool) {
        self.workers[staged.worker].buffer = Some(staged.bytes);
        if used {
            self.counters.used.fetch_add(1, Ordering::Relaxed);
        } else {
            self.counters.discarded.fetch_add(1, Ordering::Relaxed);
        }
    }
}

impl Drop for ReaderPool {
    fn drop(&mut self) {
        for worker in &mut self.workers {
            worker.tx.take();
        }
        for worker in &mut self.workers {
            if let Some(thread) = worker.thread.take() {
                let _ = thread.join();
            }
        }
    }
}

#[cfg(test)]
mod tests {
    use super::*;
    use anyhow::Context;
    use std::{
        fs,
        time::{Duration, Instant},
    };

    #[test]
    fn staging_is_bounded_exact_and_recovers_after_failed_or_wrong_predictions() -> Result<()> {
        let temp = tempfile::tempdir()?;
        let bytes: Vec<_> = (0..EXPERT_BYTES as usize)
            .map(|i| (i.wrapping_mul(31) % 251) as u8)
            .collect();
        let header = format!(
            "{{\"record\":{{\"dtype\":\"U8\",\"shape\":[{}],\"data_offsets\":[0,{}]}}}}",
            bytes.len(),
            bytes.len()
        );
        let file = temp.path().join("experts.safetensors");
        let mut payload = (header.len() as u64).to_le_bytes().to_vec();
        payload.extend(header.bytes());
        payload.extend(&bytes);
        fs::write(&file, payload)?;
        let record = crate::checkpoint::read_header(&file)?
            .remove("record")
            .context("missing fixture")?;
        let mut pool = ReaderPool::new()?;
        assert!(pool.take(0, &[0]).is_empty());
        for expert in 0..10 {
            pool.submit(0, expert, record.clone());
        }
        pool.submit(0, 0, record.clone());
        assert_eq!(pool.counters.submitted.load(Ordering::Relaxed), 4);
        pool.wait_for_test()?;
        let deadline = Instant::now() + Duration::from_secs(5);
        let mut seen = Vec::new();
        while seen.len() < 4 {
            ensure!(Instant::now() < deadline, "staging completion timeout");
            for staged in pool.take(0, &[0, 1, 2, 3]) {
                assert_eq!(staged.bytes, bytes);
                seen.push(staged.expert);
                pool.recycle(staged, true);
            }
            thread::yield_now();
        }
        seen.sort_unstable();
        assert_eq!(seen, [0, 1, 2, 3]);
        assert_eq!(
            pool.counters.bytes.load(Ordering::Relaxed),
            4 * EXPERT_BYTES
        );
        let mut invalid = record.clone();
        invalid.data_offsets = [u64::MAX - 1, u64::MAX];
        pool.submit(1, 1, invalid);
        pool.submit(1, 2, record.clone());
        pool.wait_for_test()?;
        while pool.counters.discarded.load(Ordering::Relaxed) < 2 {
            ensure!(
                Instant::now() < deadline,
                "failed staging completion timeout"
            );
            assert!(pool.take(9, &[1, 2]).is_empty());
            thread::yield_now();
        }
        assert_eq!(pool.counters.failed.load(Ordering::Relaxed), 1);
        pool.submit(3, 511, record.clone());
        let counters = Arc::clone(&pool.counters);
        drop(pool);
        assert_eq!(
            counters.completed.load(Ordering::Relaxed),
            counters.submitted.load(Ordering::Relaxed)
        );
        assert_eq!(counters.used.load(Ordering::Relaxed), 4);
        Ok(())
    }

    #[cfg(kani)]
    #[kani::proof]
    fn actual_worker_pool_bounds_and_empty_poll() {
        let mut pool = ReaderPool::new().unwrap();
        let layer = kani::any::<usize>();
        let expert = kani::any::<u32>();
        assert!(pool.take(layer, &[expert]).is_empty());
        assert_eq!(pool.workers.len(), 4);
        assert!(
            pool.workers
                .iter()
                .all(|w| w.buffer.as_ref().unwrap().len() == EXPERT_BYTES as usize)
        );
        kani::cover!(expert >= 512);
    }
}
