//! Exact routed EXL3 experts, read directly into reusable slot-major banks.
use crate::{
    array::{Array, Dtype},
    checkpoint::{Checkpoint, TensorInfo},
    codec::Codebook,
    flash_support::{EXPERT_BYTES, ExpertCache, ExpertFill},
    lfm2::Projection,
    linear, router,
    ssd_prefetch::{PREDICTION_RANKS, ReaderPool, order_predictions},
};
use anyhow::{Context, Result, ensure};
use std::{cell::RefCell, collections::BTreeSet, rc::Rc};

const H: i32 = 2560;
const I: i32 = 640;
const TRELLIS: usize = 409600;

struct ResidentHot {
    input: Array,
    output: Array,
    map: [u32; 512],
    complete: bool,
}

struct PreparedRows {
    selected: Array,
    weights: Array,
    ids: Vec<u32>,
    hot: Option<ResidentHot>,
}

pub struct SsdMoe {
    gate: Projection,
    resident_gate: Option<Array>,
    shared_gate: Projection,
    shared_up: Projection,
    shared_down: Projection,
    multiplier: Projection,
    records: Vec<TensorInfo>,
    banks: Vec<Array>,
    cache: Rc<RefCell<ExpertCache>>,
    last: Option<Array>,
    pub hits: u64,
    pub misses: u64,
    pub read_bytes: u64,
    overlap: bool,
    readers: usize,
    gpu_hot: bool,
    gpu_hot_wide: bool,
    unique_cohorts: bool,
    shared_exl3: bool,
    stable_expert_kernels: bool,
    shared_batch: bool,
    pub(crate) prefetch_pipeline: bool,
    #[cfg(test)]
    cohort_sizes: Vec<usize>,
    #[cfg(test)]
    route_readbacks: usize,
    #[cfg(test)]
    window_hot_before_readback: Vec<bool>,
    #[cfg(test)]
    hybrid_submissions: u64,
    #[cfg(test)]
    slow_hot_for_test: bool,
    #[cfg(test)]
    hot_ready_at_routes: Option<bool>,
    #[cfg(test)]
    hot_ready_at_demand: Option<bool>,

}

impl SsdMoe {
    pub fn load(c: &Checkpoint, prefix: &str, slots: usize, overlap: bool) -> Result<Self> {
        let control = |name, default: &str| -> Result<String> {
            match std::env::var(name) {
                Ok(value) => Ok(value),
                Err(std::env::VarError::NotPresent) => Ok(default.into()),
                Err(error) => Err(error.into()),
            }
        };
        let force_lru =
            crate::flash_support::parse_lru_override(&control("MLXL3_SSD_CACHE_POLICY", "auto")?)
                .context("MLXL3_SSD_CACHE_POLICY must be auto or lru")?;
        let predictor_half = crate::flash_support::parse_predictor_half(&control(
            "MLXL3_SSD_PREDICTOR_DTYPE",
            "f32",
        )?)
        .context("MLXL3_SSD_PREDICTOR_DTYPE must be f32 or f16")?;
        let resident_gates = match control("MLXL3_SSD_RESIDENT_GATES", "0")?.as_str() {
            "0" => false,
            "1" => true,
            _ => anyhow::bail!("MLXL3_SSD_RESIDENT_GATES must be 0 or 1"),
        };
        let gpu_hot = match control("MLXL3_SSD_GPU_HOT", "0")?.as_str() {
            "0" => false,
            "1" => prefix.starts_with("model.language_model.layers."),
            _ => anyhow::bail!("MLXL3_SSD_GPU_HOT must be 0 or 1"),
        };
        let gpu_hot_wide =
            crate::flash_support::parse_gpu_hot_wide(&control("MLXL3_SSD_GPU_HOT_WIDE", "0")?)
                .context("MLXL3_SSD_GPU_HOT_WIDE must be 0 or 1")?;
        let unique_cohorts =
            crate::flash_support::parse_gpu_hot_wide(&control("MLXL3_SSD_UNIQUE_COHORT", "0")?)
                .context("MLXL3_SSD_UNIQUE_COHORT must be 0 or 1")?;
        let shared_exl3 =
            crate::flash_support::parse_gpu_hot_wide(&control("MLXL3_SSD_SHARED_EXL3", "0")?)
                .context("MLXL3_SSD_SHARED_EXL3 must be 0 or 1")?;
        let stable_expert_kernels = crate::flash_support::parse_gpu_hot_wide(&control(
            "MLXL3_SSD_STABLE_EXPERT_KERNELS",
            "0",
        )?)
        .context("MLXL3_SSD_STABLE_EXPERT_KERNELS must be 0 or 1")?;
        let shared_batch =
            crate::flash_support::parse_gpu_hot_wide(&control("MLXL3_SSD_SHARED_BATCH", "0")?)
                .context("MLXL3_SSD_SHARED_BATCH must be 0 or 1")?;
        let prefetch_pipeline =
            crate::flash_support::parse_gpu_hot_wide(&control("MLXL3_SSD_PREFETCH_PIPELINE", "0")?)
                .context("MLXL3_SSD_PREFETCH_PIPELINE must be 0 or 1")?;
        let readers = std::env::var("MLXL3_SSD_READERS")
            .unwrap_or_else(|_| "4".into())
            .parse::<usize>()
            .context("invalid SSD reader count")?;
        ensure!((1..=4).contains(&readers), "SSD readers must be in 1..=4");
        let mut records = Vec::with_capacity(512);
        for expert in 0..512 {
            let p = format!("{prefix}.experts.{expert}");
            let mut start = None;
            let mut cursor = 0;
            let mut file = None;
            for (projection, input, output) in
                [("down_proj", I, H), ("gate_proj", H, I), ("up_proj", H, I)]
            {
                for (part, shape, dtype) in [
                    ("suh", vec![input as usize], "F16"),
                    ("svh", vec![output as usize], "F16"),
                    ("mul1", vec![], "I32"),
                    (
                        "trellis",
                        vec![input as usize / 16, output as usize / 16, 32],
                        "I16",
                    ),
                ] {
                    let name = format!("{p}.{projection}.{part}");
                    let tensor = c
                        .tensors
                        .get(&name)
                        .with_context(|| format!("missing {name}"))?;
                    ensure!(
                        tensor.shape == shape && tensor.dtype == dtype,
                        "unsupported Flash expert tensor {name}"
                    );
                    if start.is_none() {
                        start = Some(tensor.clone());
                        cursor = tensor.data_offsets[0];
                        file = Some(tensor.file.clone());
                    }
                    ensure!(
                        Some(&tensor.file) == file.as_ref() && tensor.data_offsets[0] == cursor,
                        "Flash expert record is not contiguous"
                    );
                    cursor = tensor.data_offsets[1];
                }
            }
            let first = start.context("empty expert record")?;
            ensure!(
                cursor - first.data_offsets[0] == EXPERT_BYTES,
                "invalid Flash expert record size"
            );
            records.push(first);
        }
        let s = i32::try_from(slots)?;
        let cache = Rc::new(RefCell::new(ExpertCache::new(slots)?));
        if force_lru {
            cache.borrow_mut().enable_predictive_lru();
        }
        let banks = [
            (vec![s, 2, 160, 40, 32], Dtype::UInt16), // gate/up trellises
            (vec![s, 1, 40, 160, 32], Dtype::UInt16), // down trellis
            (vec![s, 2, H], Dtype::Float16),
            (vec![s, 2, I], Dtype::Float16),
            (vec![s, 1, I], Dtype::Float16),
            (vec![s, 1, H], Dtype::Float16),
            (vec![s, 3], Dtype::Int32),
        ]
        .into_iter()
        .map(|(shape, dtype)| Array::zeros_dtype(&shape, dtype))
        .collect::<Result<Vec<_>>>()?;
        for bank in &banks {
            bank.eval()?;
        }
        let projection = |suffix: &str, input, output| {
            Projection::load(c, &format!("{prefix}.{suffix}"), input, output, false)
        };
        let resident_gate = if resident_gates && prefix.starts_with("model.language_model.layers.")
        {
            let name = format!("{prefix}.gate.weight");
            let info = c.tensors.get(&name).context("missing dense Flash gate")?;
            ensure!(
                info.shape == [512, H as usize]
                    && matches!(info.dtype.as_str(), "F16" | "BF16" | "F32")
                    && !c.tensors.contains_key(&format!("{prefix}.gate.bias")),
                "resident Flash gate must be dense 512x2560 without bias"
            );
            // Match MLX matmul's transpose-then-promote order, once at load.
            let weight = linear::checkpoint_array(c, &name)?
                .astype(Dtype::Float16)?
                .transpose(&[1, 0])?
                .astype(Dtype::Float32)?;
            weight.eval()?;
            Some(weight)
        } else {
            None
        };
        Ok(Self {
            gpu_hot,
            gpu_hot_wide,
            unique_cohorts,
            shared_exl3,
            stable_expert_kernels,
            shared_batch,
            prefetch_pipeline,
            #[cfg(test)]
            cohort_sizes: Vec::new(),
            #[cfg(test)]
            route_readbacks: 0,
            #[cfg(test)]
            window_hot_before_readback: Vec::new(),
            #[cfg(test)]
            hybrid_submissions: 0,
            #[cfg(test)]
            slow_hot_for_test: false,
            #[cfg(test)]
            hot_ready_at_routes: None,
            #[cfg(test)]
            hot_ready_at_demand: None,
            gate: projection("gate", H, 512)?,
            resident_gate,
            shared_gate: projection("shared_expert.gate_proj", H, I)?,
            shared_up: projection("shared_expert.up_proj", H, I)?,
            shared_down: projection("shared_expert.down_proj", I, H)?,
            multiplier: projection("shared_expert_gate", H, 1)?,
            records,
            banks,
            cache,
            last: None,
            hits: 0,
            misses: 0,
            read_bytes: 0,
            overlap,
            readers,
            predictor_half,
            prefetcher: None,
            #[cfg(test)]
            route_ids: Vec::new(),
            #[cfg(test)]
            profile_seconds: [0.0; 2],
            #[cfg(test)]
            prefetched: 0,
            #[cfg(test)]
            prefetch_profile: [0; 4],
            #[cfg(test)]
            overlap_hot_rows: 0,
            #[cfg(test)]
            import_seconds: 0.0,
            #[cfg(test)]
            schedule_events: Vec::with_capacity(8),
            #[cfg(test)]
            prediction_ready_at_demand: None,
            #[cfg(test)]
            slow_prediction_for_test: false,
        })
    }

    fn destinations(slot: usize) -> [(usize, usize, usize); 12] {
        [
            (4, slot * I as usize * 2, I as usize * 2),
            (5, slot * H as usize * 2, H as usize * 2),
            (6, slot * 12, 4),
            (1, slot * TRELLIS, TRELLIS),
            (2, slot * 2 * H as usize * 2, H as usize * 2),
            (3, slot * 2 * I as usize * 2, I as usize * 2),
            (6, slot * 12 + 4, 4),
            (0, slot * 2 * TRELLIS, TRELLIS),
            (2, (slot * 2 + 1) * H as usize * 2, H as usize * 2),
            (3, (slot * 2 + 1) * I as usize * 2, I as usize * 2),
            (6, slot * 12 + 8, 4),
            (0, (slot * 2 + 1) * TRELLIS, TRELLIS),
        ]
    }

    fn fill_many(&mut self, jobs: &[(u32, usize)]) -> Result<()> {
        if jobs.is_empty() {
            return Ok(());
        }
        #[cfg(test)]
        let _profile = crate::flash_profile::span("fill_prepare_and_read");
        let sources = jobs
            .iter()
            .map(|&(expert, _)| {
                self.records
                    .get(expert as usize)
                    .context("invalid expert")?
                    .source()
            })
            .collect::<Result<Vec<_>>>()?;
        let destinations = jobs
            .iter()
            .map(|&(_, slot)| Self::destinations(slot))
            .collect::<Vec<_>>();
        let views = destinations
            .iter()
            .map(|ranges| ranges.as_slice())
            .collect::<Vec<_>>();
        let bytes = EXPERT_BYTES
            .checked_mul(jobs.len() as u64)
            .and_then(|n| self.read_bytes.checked_add(n))
            .context("SSD byte counter exhausted")?;
        #[cfg(test)]
        let read_profile = crate::flash_profile::span("disk_read_batch");
        let result = Array::read_records(&sources, &mut self.banks, &views, self.readers);
        #[cfg(test)]
        if let Some(profile) = &read_profile {
            profile.data(serde_json::json!({"jobs":jobs,"workers":self.readers,"timings_job_worker_start_end_bytes_failed":crate::array::disk_profile()?,"ok":result.is_ok()}));
        }
        result?;
        self.read_bytes = bytes;
        Ok(())
    }

    fn calculate(&self, x: &Array, slots: &[u32]) -> Result<Array> {
        let capacity = self.banks[0].shape()[0];
        let slots = if self.shared_exl3 {
            linear::ExpertSlots::grouped(slots, capacity)?
        } else {
            linear::ExpertSlots::new(slots, capacity)?
        };
        self.calculate_slots(x, &slots)
    }

    fn calculate_slots(&self, x: &Array, slots: &linear::ExpertSlots) -> Result<Array> {
        let n = slots.indices().shape()[0];
        #[cfg(test)]
        let profile = crate::flash_profile::span("expert_calculate_build");
        #[cfg(test)]
        if let Some(profile) = &profile {
            profile.data(serde_json::json!({"rows":n}));
        }
        ensure!(n > 0 && x.shape() == [n, H], "invalid expert slot mapping");
        let expanded = x
            .reshape(&[n, 1, H])?
            .broadcast_to(&[n, 2, H])?
            .reshape(&[n * 2, H])?;
        let suh = slots.gather_scales(&self.banks[2])?.reshape(&[n * 2, H])?;
        let svh = slots.gather_scales(&self.banks[3])?.reshape(&[n * 2, I])?;
        let gu = linear::expert_slots(
            &expanded,
            &self.banks[0],
            &suh,
            &svh,
            slots,
            I,
            2,
            2,
            Codebook::Mul1,
            self.stable_expert_kernels,
        )?
        .reshape(&[n, 2, I])?;
        #[cfg(test)]
        crate::flash_profile::stage("expert_gate_up_completion", &gu)?;
        let activated = gu
            .slice(1, 0, 1)?
            .reshape(&[n, I])?
            .swiglu(&gu.slice(1, 1, 2)?.reshape(&[n, I])?)?;
        #[cfg(test)]
        crate::flash_profile::stage("expert_swiglu_completion", &activated)?;
        let down = linear::expert_slots(
            &activated,
            &self.banks[1],
            &slots.gather_scales(&self.banks[4])?.reshape(&[n, I])?,
            &slots.gather_scales(&self.banks[5])?.reshape(&[n, H])?,
            slots,
            H,
            1,
            2,
            Codebook::Mul1,
            self.stable_expert_kernels,
        )?;
        #[cfg(test)]
        crate::flash_profile::stage("expert_down_completion", &down)?;
        Ok(down)
    }

    fn gate_logits(&self, x: &Array) -> Result<Array> {
        match &self.resident_gate {
            Some(weight) if x.dtype() == Dtype::Float32 => x.matmul(weight),
            _ => self.gate.forward(x),
        }
    }

    pub(crate) fn route(&self, x: &Array) -> Result<(Array, Array)> {
        ensure!(
            x.shape() == [1, H] && matches!(x.dtype(), Dtype::Float16 | Dtype::Float32),
            "streamed MoE expects one exact route row"
        );
        let probabilities = self.gate_logits(x)?.softmax_precise()?;
        if x.dtype() == Dtype::Float16 {
            let (indices, scores) = router::topk(&probabilities, 10, false)?;
            Ok((indices, scores.div(&scores.sum(1, true)?)?))
        } else {
            router::topk(&probabilities, 10, true)
        }
    }
    fn predict(&self, x: &Array) -> Result<Array> {
        let x = if self.predictor_half {
            x.astype(Dtype::Float16)?
        } else {
            x.try_clone()?
        };
        let scores = self.gate.forward(&x)?;
        #[cfg(test)]
        let scores = if self.slow_prediction_for_test {
            crate::array::metal_kernel(
                "mlxl3_test_slow_prediction",
                &["scores"], &["out"], "",
                "uint i=thread_position_in_grid.x; float v=float(scores[i]); for(uint j=0;j<1000000u;++j) v=fma(v,0.99999f,0.0001f); out[i]=v;",
                &[&scores], &[scores.shape().to_vec()], &[Dtype::Float32],
                [512, 1, 1], [128, 1, 1],
            )?.remove(0)
        } else {
            scores
        };
        scores
            .argsort()?
            .slice(1, 512 - PREDICTION_RANKS as i32, 512)
    }
    pub fn forward(&mut self, x: &Array) -> Result<Array> {
        self.forward_many(x)
    }

    pub(crate) fn warm_all(&mut self) -> Result<()> {
        self.drain()?;
        ensure!(
            self.cache.borrow().capacity() == 512,
            "resident draft requires all expert slots"
        );
        let ids = (0..512u32).collect::<Vec<_>>();
        let routes = self.cache.borrow_mut().reserve(&ids)?;
        let jobs = ids
            .iter()
            .zip(&routes)
            .filter_map(|(&id, &(slot, hit))| (!hit).then_some((id, slot)))
            .collect::<Vec<_>>();
        if let Err(error) = self.fill_many(&jobs) {
            self.cache.borrow_mut().invalidate();
            return Err(error);
        }
        Ok(())
    }

    pub(crate) fn forward_many(&mut self, x: &Array) -> Result<Array> {
        self.forward_lookahead(x, None, 0)
    }

    pub(crate) fn forward_lookahead(
        &mut self,
        x: &Array,
        next: Option<&mut Self>,
        depth: usize,
    ) -> Result<Array> {
        self.forward_rows(x, next, depth, 2)
    }

    #[cfg(any(feature = "chat", test))]
    pub(crate) fn forward_draft_rows(&mut self, x: &Array) -> Result<Array> {
        self.forward_rows(x, None, 0, 8)
    }

    #[cfg(any(feature = "chat", test))]
    pub(crate) fn forward_draft_lookahead(
        &mut self,
        x: &Array,
        next: Option<&mut Self>,
        depth: usize,
    ) -> Result<Array> {
        if self.prefetch_pipeline || depth == 0 {
            return self.forward_draft_rows(x);
        }
        ensure!(
            depth <= 10 && x.shape().len() == 2,
            "invalid draft lookahead"
        );
        let rows = crate::ssd_prefetch::prediction_rows(usize::try_from(x.shape()[0])?)
            .context("invalid draft prediction rows")?;
        let prediction = next
            .as_ref()
            .map(|next| next.predict(&x.slice(0, 0, rows as i32)?))
            .transpose()?;
        if let Some(prediction) = &prediction {
            prediction.async_eval()?;
        }
        let output = self.forward_draft_rows(x)?;
        if let (Some(next), Some(prediction)) = (next, prediction) {
            output.async_eval()?;
            let mut ids = prediction.to_u32()?;
            ensure!(
                order_predictions(&mut ids, rows),
                "invalid draft predictions"
            );
            next.prefetch(&ids, depth)?;
        }
        Ok(output)
    }

    fn forward_rows(
        &mut self,
        x: &Array,
        next: Option<&mut Self>,
        depth: usize,
        row_limit: i32,
    ) -> Result<Array> {
        self.forward_rows_prepared(x, next, depth, row_limit, None)
    }

    fn routed_input(x: &Array) -> Result<Array> {
        let rows = x.shape()[0];
        let indices = (0..rows as u32)
            .flat_map(|row| std::iter::repeat_n(row, 10))
            .collect::<Vec<_>>();
        x.take(&Array::from_u32(&indices, &[rows * 10])?, 0)?
            .astype(Dtype::Float16)
    }

    fn prepare_resident_hot(&mut self, x: &Array, selected: &Array) -> Result<Option<ResidentHot>> {
        if let Some(last) = self.last.take() {
            last.eval()?;
        }
        let map = self.cache.borrow().resident_slot_map()?;
        if map.iter().all(|&slot| slot == u32::MAX) {
            return Ok(None);
        }
        let input = Self::routed_input(x)?;
        #[cfg(test)]
        let _prepare = crate::flash_profile::span("gpu_resident_hot_prepare");
        let slots = linear::ExpertSlots::residents(selected, &map, self.banks[0].shape()[0])?;
        slots.indices().async_eval()?;
        let output = self.calculate_slots(&input, &slots)?;
        #[cfg(test)]
        let output = if self.slow_hot_for_test {
            hybrid_tests::delay(&output)?
        } else {
            output
        };
        self.last = Some(output.try_clone()?);
        output.async_eval()?;
        #[cfg(test)]
        {
            self.hybrid_submissions += 1;
        }
        Ok(Some(ResidentHot {
            input,
            output,
            map,
            complete: false,
        }))
    }

    fn forward_unique_rows(&mut self, x: &Array, row_limit: i32) -> Result<Array> {
        #[cfg(test)]
        let _cohorts = crate::flash_profile::span("moe_unique_cohorts");
        #[cfg(test)]
        let planning = crate::flash_profile::span("route_window_build_and_wait_inclusive");
        #[cfg(test)]
        let started = std::time::Instant::now();
        #[cfg(test)]
        let submissions_before = self.hybrid_submissions;
        let rows = x.shape()[0];
        let pairs = (0..rows)
            .map(|row| self.route(&x.slice(0, row, row + 1)?))
            .collect::<Result<Vec<_>>>()?;
        let selected = Array::concatenate(&pairs.iter().map(|p| &p.0).collect::<Vec<_>>(), 0)?;
        let weights = Array::concatenate(&pairs.iter().map(|p| &p.1).collect::<Vec<_>>(), 0)?;
        let mut hot = self.prepare_resident_hot(x, &selected)?;
        #[cfg(test)]
        self.window_hot_before_readback
            .push(self.hybrid_submissions > submissions_before);
        let ids = selected.to_u32()?;
        ensure!(ids.len() == rows as usize * 10, "incomplete route window");
        #[cfg(test)]
        {
            self.route_readbacks += 1;
            self.profile_seconds[0] += started.elapsed().as_secs_f64();
            drop(planning);
        }
        let available = self.cache.borrow().available_slots();
        if crate::flash_support::largest_expert_prefix(&ids, available) == Some(rows as usize) {
            return self.forward_rows_prepared(
                x,
                None,
                0,
                row_limit,
                Some(PreparedRows {
                    selected,
                    weights,
                    ids,
                    hot,
                }),
            );
        }
        // A split may evict future Hot experts: finish their captured results first.
        if let Some(hot) = &mut hot {
            let hot_ids = ids
                .iter()
                .copied()
                .filter(|&id| {
                    hot.map
                        .get(id as usize)
                        .is_some_and(|&slot| slot != u32::MAX)
                })
                .collect::<Vec<_>>();
            // Reused Hot values still count as demand before Cold chooses its victims.
            if !hot_ids.is_empty() {
                ensure!(
                    self.cache
                        .borrow_mut()
                        .reserve(&hot_ids)?
                        .iter()
                        .all(|p| p.1),
                    "resident Hot snapshot changed before reservation"
                );
            }
            hot.output.eval()?;
            hot.complete = true;
            self.last = None;
        }
        let mut output = Vec::new();
        let mut begin = 0;
        while begin < rows {
            let available = self.cache.borrow().available_slots();
            let count =
                crate::flash_support::largest_expert_prefix(&ids[begin as usize * 10..], available)
                    .filter(|&count| count > 0)
                    .context("route window has no reservable expert prefix")?
                    as i32;
            let end = begin + count;
            if let Some(last) = self.last.take() {
                last.eval()?;
            }
            output.push(
                self.forward_rows_prepared(
                    &x.slice(0, begin, end)?,
                    None,
                    0,
                    row_limit,
                    Some(PreparedRows {
                        selected: selected.slice(0, begin, end)?,
                        weights: weights.slice(0, begin, end)?,
                        ids: ids[begin as usize * 10..end as usize * 10].to_vec(),
                        hot: hot
                            .as_ref()
                            .map(|hot| {
                                Ok::<_, anyhow::Error>(ResidentHot {
                                    input: hot.input.slice(0, begin * 10, end * 10)?,
                                    output: hot.output.slice(0, begin * 10, end * 10)?,
                                    map: hot.map,
                                    complete: true,
                                })
                            })
                            .transpose()?,
                    }),
                )?,
            );
            begin = end;
        }
        Array::concatenate(&output.iter().collect::<Vec<_>>(), 0)
    }

    fn forward_rows_prepared(
        &mut self,
        x: &Array,
        next: Option<&mut Self>,
        depth: usize,
        row_limit: i32,
        prepared: Option<PreparedRows>,
    ) -> Result<Array> {
        ensure!(depth <= 10, "lookahead depth must be in 0..=10");
        ensure!(
            x.shape().len() == 2
                && (1..=row_limit).contains(&x.shape()[0])
                && x.shape()[1] == H
                && matches!(x.dtype(), Dtype::Float16 | Dtype::Float32),
            "invalid streamed MoE route shape or dtype"
        );
        let rows = x.shape()[0];
        if let Some((_, pool)) = &self.prefetcher {
            pool.borrow_mut().poll();
        }
        let available = self.cache.borrow().available_slots();
        if prepared.is_none() && available < rows as usize * 10 {
            if self.unique_cohorts
                && self.gpu_hot
                && self.gpu_hot_wide
                && self.overlap
                && depth == 0
                && (self.prefetcher.is_none() || self.prefetch_pipeline)
            {
                return self.forward_unique_rows(x, row_limit);
            }
            #[cfg(test)]
            let _cohorts = crate::flash_profile::span("moe_cohorts");
            let cohort = crate::flash_support::expert_cohort_rows(available)
                .context("invalid streamed MoE cache capacity")? as i32;
            let mut output = Vec::new();
            for begin in (0..rows).step_by(cohort as usize) {
                output.push(self.forward_rows(
                    &x.slice(0, begin, (begin + cohort).min(rows))?,
                    None,
                    0,
                    row_limit,
                )?);
            }
            return Array::concatenate(&output.iter().collect::<Vec<_>>(), 0);
        }
        #[cfg(test)]
        self.cohort_sizes.push(rows as usize);
        #[cfg(test)]
        let profile = crate::flash_profile::span("moe_forward_rows");
        #[cfg(test)]
        let route_start = std::time::Instant::now();
        #[cfg(test)]
        {
            self.schedule_events.clear();
            self.prediction_ready_at_demand = None;
            self.hot_ready_at_routes = None;
            self.hot_ready_at_demand = None;
        }
        #[cfg(test)]
        let routing = crate::flash_profile::span("router_build_and_wait_inclusive");
        let hybrid = self.gpu_hot
            && self.overlap
            && (rows <= 2 || self.gpu_hot_wide)
            && depth == 0
            && (self.prefetcher.is_none() || self.prefetch_pipeline);
        let (selected, weights, host_ids, hot) = match prepared {
            Some(prepared) => (
                prepared.selected,
                prepared.weights,
                Some(prepared.ids),
                prepared.hot,
            ),
            None => {
                let pairs = (0..rows)
                    .map(|row| self.route(&x.slice(0, row, row + 1)?))
                    .collect::<Result<Vec<_>>>()?;
                let selected =
                    Array::concatenate(&pairs.iter().map(|p| &p.0).collect::<Vec<_>>(), 0)?;
                let weights =
                    Array::concatenate(&pairs.iter().map(|p| &p.1).collect::<Vec<_>>(), 0)?;
                let hot = if hybrid {
                    self.prepare_resident_hot(x, &selected)?
                } else {
                    None
                };
                (selected, weights, None, hot)
            }
        };
        let (early_input, gpu_hot, snapshot, completed_hot) = match hot {
            Some(hot) => (
                Some(hot.input),
                Some(hot.output),
                Some(hot.map),
                hot.complete,
            ),
            None => (None, None, None, false),
        };
        let early_hot = gpu_hot.is_some();
        #[cfg(test)]
        let readback = crate::flash_profile::span("route_readback_inclusive");
        let ids = match host_ids {
            Some(ids) => ids,
            None => {
                #[cfg(test)]
                {
                    self.route_readbacks += 1;
                }
                selected.to_u32()?
            }
        };
        #[cfg(test)]
        {
            drop(readback);
            drop(routing);
        }
        #[cfg(test)]
        {
            self.profile_seconds[0] += route_start.elapsed().as_secs_f64();
            self.schedule_events
                .push(("routes_cpu", route_start.elapsed().as_secs_f64()));
        }
        #[cfg(test)]
        {
            self.route_ids.clone_from(&ids);
            self.hot_ready_at_routes = gpu_hot
                .as_ref()
                .map(Array::is_available_for_test)
                .transpose()?;
        }
        let routed_x = match early_input {
            Some(value) => value,
            None => Self::routed_input(x)?,
        };
        if !early_hot && let Some(last) = self.last.take() {
            #[cfg(test)]
            let _profile = crate::flash_profile::span("previous_bank_users_wait");
            last.eval()?;
        }
        #[cfg(test)]
        let import_start = std::time::Instant::now();
        if let Some((layer, pool)) = &self.prefetcher {
            self.cache.borrow_mut().enable_predictive_lru();
            let mut pool = pool.borrow_mut();
            pool.poll();
            pool.note_late(*layer, &ids);
            if self.prefetch_pipeline {
                pool.wait_demand(*layer, &self.cache, &ids);
            }
        }
        let before = self.cache.borrow().prefetch_hits;
        let fill = ExpertFill::new(&self.cache);
        let snapshot_hot = |id: u32| {
            snapshot
                .as_ref()
                .and_then(|map| map.get(id as usize))
                .is_some_and(|&slot| slot != u32::MAX)
        };
        let routes = if completed_hot {
            // Completed Hot values survive eviction; reserve only the original Cold IDs.
            let cold_ids = ids
                .iter()
                .copied()
                .filter(|&id| !snapshot_hot(id))
                .collect::<Vec<_>>();
            let reserved = if cold_ids.is_empty() {
                Vec::new()
            } else {
                self.cache.borrow_mut().reserve(&cold_ids)?
            };
            let mut reserved = reserved.into_iter();
            ids.iter()
                .map(|&id| {
                    if snapshot_hot(id) {
                        (
                            snapshot.as_ref().expect("Hot snapshot")[id as usize] as usize,
                            true,
                        )
                    } else {
                        reserved.next().expect("reserved Cold occurrence")
                    }
                })
                .collect::<Vec<_>>()
        } else {
            self.cache.borrow_mut().reserve(&ids)?
        };
        let used = self.cache.borrow().prefetch_hits - before;
        if let Some((_, pool)) = &self.prefetcher {
            pool.borrow()
                .counters
                .used
                .fetch_add(used, std::sync::atomic::Ordering::Relaxed);
        }
        #[cfg(test)]
        {
            self.prefetched += used;
            self.import_seconds += import_start.elapsed().as_secs_f64();
        }
        let mut hot = Vec::new();
        let mut cold = Vec::new();
        let mut order = Vec::new();
        for (i, &(slot, hit)) in routes.iter().enumerate() {
            if hit {
                self.hits = self.hits.saturating_add(1);
            } else {
                self.misses = self.misses.saturating_add(1);
            }
            if if early_hot { snapshot_hot(ids[i]) } else { hit } {
                hot.push((i, slot as u32));
            } else {
                cold.push((i, slot as u32));
            }
        }
        #[cfg(test)]
        if let Some(profile) = &profile {
            let hits = routes.iter().filter(|p| p.1).count();
            profile.data(serde_json::json!({"rows":rows,"ids":ids,"slots_hit":routes,"hot_rows":hits,"cold_rows":ids.len()-hits,"resident_gpu_rows":hot.len(),"completed_hot":completed_hot,"capacity":self.cache.borrow().capacity()}));
        }
        let hot_output = if early_hot {
            #[cfg(test)]
            {
                self.overlap_hot_rows += hot.len() as u64;
            }
            gpu_hot
        } else if self.overlap && !hot.is_empty() && !cold.is_empty() {
            let value = self.calculate(
                &routed_x.take(
                    &Array::from_u32(
                        &hot.iter().map(|p| p.0 as u32).collect::<Vec<_>>(),
                        &[hot.len() as i32],
                    )?,
                    0,
                )?,
                &hot.iter().map(|p| p.1).collect::<Vec<_>>(),
            )?;
            // Retain submitted reads of the banks even if a later operation fails.
            self.last = Some(value.try_clone()?);
            value.async_eval()?;
            #[cfg(test)]
            {
                self.overlap_hot_rows += hot.len() as u64;
            }
            Some(value)
        } else {
            None
        };
        let prediction = (|| -> Result<Option<Array>> {
            let prediction = if depth > 0 {
                next.as_ref().map(|next| next.predict(x)).transpose()?
            } else {
                None
            };
            if let Some(prediction) = &prediction {
                prediction.async_eval()?;
                #[cfg(test)]
                self.schedule_events
                    .push(("prediction_submitted", route_start.elapsed().as_secs_f64()));
            }
            Ok(prediction)
        })();
        let prediction = match prediction {
            Ok(prediction) => prediction,
            Err(error) => {
                if let Some(hot) = &hot_output {
                    hot.eval()?;
                }
                return Err(error.context("preparing optional expert prediction"));
            }
        };
        let mut filled = BTreeSet::new();
        #[cfg(test)]
        let read_start = std::time::Instant::now();
        let jobs = cold
            .iter()
            .filter(|&&(i, _)| !routes[i].1)
            .filter_map(|&(i, slot)| filled.insert(slot).then_some((ids[i], slot as usize)))
            .collect::<Vec<_>>();
        #[cfg(test)]
        if !jobs.is_empty() {
            self.hot_ready_at_demand = hot_output
                .as_ref()
                .map(Array::is_available_for_test)
                .transpose()?;
            self.prediction_ready_at_demand = prediction
                .as_ref()
                .map(Array::is_available_for_test)
                .transpose()?;
            self.schedule_events
                .push(("demand_begin", route_start.elapsed().as_secs_f64()));
        }
        if let Err(error) = self.fill_many(&jobs) {
            if let Some(hot) = &hot_output {
                hot.eval()?;
            }
            return Err(error.context("loading cold EXL3 experts"));
        }
        fill.commit();
        #[cfg(test)]
        {
            self.profile_seconds[1] += read_start.elapsed().as_secs_f64();
            if !jobs.is_empty() {
                self.schedule_events
                    .push(("demand_end", route_start.elapsed().as_secs_f64()));
            }
        }
        let values = if let Some(hot_output) = hot_output {
            if cold.is_empty() {
                hot_output
            } else {
                let cold_output = self.calculate(
                    &routed_x.take(
                        &Array::from_u32(
                            &cold.iter().map(|p| p.0 as u32).collect::<Vec<_>>(),
                            &[cold.len() as i32],
                        )?,
                        0,
                    )?,
                    &cold.iter().map(|p| p.1).collect::<Vec<_>>(),
                )?;
                let combined = Array::concatenate(&[&hot_output, &cold_output], 0)?;
                for (i, &(_, hit)) in routes.iter().enumerate() {
                    let index = if early_hot && snapshot_hot(ids[i]) {
                        i
                    } else if !early_hot && hit {
                        hot.iter().position(|p| p.0 == i).expect("hot route")
                    } else {
                        (if early_hot { ids.len() } else { hot.len() })
                            + cold.iter().position(|p| p.0 == i).expect("cold route")
                    };
                    order.push(index as u32);
                }
                combined.take(&Array::from_u32(&order, &[order.len() as i32])?, 0)?
            }
        } else {
            self.calculate(
                &routed_x,
                &routes.iter().map(|p| p.0 as u32).collect::<Vec<_>>(),
            )?
        };
        let routed = values
            .astype(x.dtype())?
            .mul(&weights.reshape(&[rows * 10, 1])?)?
            .reshape(&[rows, 10, H])?
            .sum(1, false)?;
        #[cfg(test)]
        crate::flash_profile::stage("routed_reduction_completion", &routed)?;
        let output = routed.add(&self.shared(x)?)?;
        #[cfg(test)]
        crate::flash_profile::stage("moe_add_completion", &output)?;
        self.last = Some(output.try_clone()?);
        let predicted_ids = prediction
            .map(|p| -> Result<Vec<u32>> {
                let mut ids = p.to_u32()?;
                ensure!(
                    order_predictions(&mut ids, rows as usize),
                    "invalid lookahead prediction shape or IDs"
                );
                #[cfg(test)]
                self.schedule_events
                    .push(("prediction_cpu", route_start.elapsed().as_secs_f64()));
                Ok(ids)
            })
            .transpose()?;
        if let (Some(next), Some(ids)) = (next, predicted_ids) {
            if ids.iter().any(|&id| !next.cache.borrow().contains(id)) {
                output.async_eval()?;
            }
            next.prefetch(&ids, depth)?;
        }
        Ok(output)
    }

    fn shared_projection(&self, x: &Array) -> Result<Array> {
        let gate = self.shared_gate.forward(x)?.astype(Dtype::Float16)?;
        #[cfg(test)]
        crate::flash_profile::stage("shared_gate_projection", &gate)?;
        let up = self.shared_up.forward(x)?.astype(Dtype::Float16)?;
        #[cfg(test)]
        crate::flash_profile::stage("shared_up_projection", &up)?;
        let activated = gate.swiglu(&up)?;
        #[cfg(test)]
        crate::flash_profile::stage("shared_swiglu", &activated)?;
        let shared = self.shared_down.forward(&activated)?.astype(x.dtype())?;
        #[cfg(test)]
        crate::flash_profile::stage("shared_down_projection", &shared)?;
        Ok(shared)
    }

    fn shared(&self, x: &Array) -> Result<Array> {
        ensure!(
            x.shape().len() == 2
                && (1..=8).contains(&x.shape()[0])
                && x.shape()[1] == H
                && matches!(x.dtype(), Dtype::Float16 | Dtype::Float32),
            "invalid shared expert shape or dtype"
        );
        let rows = x.shape()[0];
        let batched = (self.shared_batch
            && rows > 1
            && self.shared_gate.is_exl3()
            && self.shared_up.is_exl3()
            && self.shared_down.is_exl3())
        .then(|| self.shared_projection(x))
        .transpose()?;
        let mut output = Vec::new();
        for row in 0..rows {
            let x = x.slice(0, row, row + 1)?;
            let shared = match &batched {
                Some(values) => values.slice(0, row, row + 1)?,
                None => self.shared_projection(&x)?,
            };
            // Keep the dense scalar GEMV's single-row reduction order.
            let shared = shared.mul(&self.multiplier.forward(&x)?.sigmoid()?)?;
            #[cfg(test)]
            crate::flash_profile::stage("shared_expert_completion", &shared)?;
            output.push(shared);
        }
        Array::concatenate(&output.iter().collect::<Vec<_>>(), 0)
    }

    fn prefetch(&mut self, ids: &[u32], budget: usize) -> Result<()> {
        ensure!(
            !ids.is_empty() && ids.len() <= 2 * PREDICTION_RANKS && ids.iter().all(|&id| id < 512),
            "invalid predicted experts"
        );
        let Some((layer, pool)) = self.prefetcher.as_ref().map(|(l, p)| (*l, Rc::clone(p))) else {
            return Ok(());
        };
        let mut pool = pool.borrow_mut();
        pool.poll();
        self.cache.borrow_mut().protect_predictions(ids)?;
        let mut cold = Vec::new();
        #[cfg(test)]
        {
            self.prefetch_profile[0] += ids.len() as u64;
        }
        for &id in ids {
            #[cfg(test)]
            {
                self.prefetch_profile[1] += 1;
            }
            if self.cache.borrow().contains(id) {
                #[cfg(test)]
                {
                    self.prefetch_profile[2] += 1;
                }
            } else if !cold.contains(&id) {
                cold.push(id);
            }
        }
        if cold.is_empty() || budget == 0 || !pool.has_room() {
            return Ok(());
        }
        self.drain()?;
        let mut admitted = 0;
        for id in cold {
            if admitted == budget.min(4) || !pool.has_room() {
                break;
            }
            let Some(slot) = self.cache.borrow_mut().claim_prefetch(id)? else {
                continue;
            };
            let read = self.records[id as usize]
                .source()
                .and_then(|(file, offset)| {
                    Array::prepare_read(file, offset, &mut self.banks, &Self::destinations(slot))
                });
            let read = match read {
                Ok(read) => read,
                Err(_) => {
                    self.cache.borrow_mut().publish_prefetch(slot, id, false);
                    pool.counters
                        .failed
                        .fetch_add(1, std::sync::atomic::Ordering::Relaxed);
                    continue;
                }
            };
            if pool.submit(layer, id, slot, Rc::clone(&self.cache), read) {
                admitted += 1;
                #[cfg(test)]
                {
                    self.prefetch_profile[3] += 1;
                }
            }
        }
        Ok(())
    }

    pub fn drain(&mut self) -> Result<()> {
        if let Some(last) = self.last.take() {
            last.eval()?;
        }
        Ok(())
    }

                                && scores.iter().all(|v| v.is_finite())
                                && (scores.iter().sum::<f32>() - 1.).abs() < 0.002
                        );
                        if layer == 0 && dtype == Dtype::Float32 && !strided {
                            let weight = moe.resident_gate.take().context("missing weight")?;
                            moe.resident_gate = Some(weight.scalar_mul(0.5)?);
                            ensure!(
                                moe.gate_logits(&x)?.to_bytes()? != original.to_bytes()?,
                                "bad gate mutation escaped oracle"
                            );
                            moe.resident_gate = Some(weight);
                            ensure!(moe.gate_logits(&x)?.to_bytes()? == original.to_bytes()?);
                        }
                        cases += 1;
                    }
                }
            }
        }
        ensure!(cases == 384);
        println!(
            "FLASH_RESIDENT_GATE cases=384 logits=196608 routes=3840 mutation=detected_restored parity=byte_exact"
        );
        Ok(())
    }

    #[test]
    #[ignore = "physical Apple GPU; experiment49 resident gate component timing only"]
    fn resident_gate_component_microbench() -> anyhow::Result<()> {
        use super::*;
        use std::{fs::File, time::Instant};
        fn select(
            models: &mut [(SsdMoe, [Array; 2])],
            saved: &mut [Option<Array>],
            candidate: bool,
        ) {
            for ((moe, _), cache) in models.iter_mut().zip(saved) {
                if candidate != moe.resident_gate.is_some() {
                    std::mem::swap(&mut moe.resident_gate, cache);
                }
                assert_eq!(moe.resident_gate.is_some(), candidate);
                assert_ne!(moe.resident_gate.is_some(), cache.is_some());
            }
        }
        fn sweep(models: &[(SsdMoe, [Array; 2])]) -> Result<Vec<u32>> {
            let mut ids = Vec::with_capacity(960);
            for (moe, inputs) in models {
                let a = moe.route(&inputs[0])?;
                let b = moe.route(&inputs[1])?;
                ids.extend(Array::concatenate(&[&a.0, &b.0], 0)?.to_u32()?);
            }
            ensure!(ids.len() == 960);
            Ok(ids)
        }
        let output = std::env::var("MLXL3_FLASH_GATE_MICRO")?;
        serde_json::to_writer(
            File::create_new(&output)?,
            &serde_json::json!({"status":"incomplete"}),
        )?;
        let mut models = resident_gate_fixture()?;
        let mut saved = (0..48).map(|_| None).collect::<Vec<_>>();
        select(&mut models, &mut saved, false);
        let oracle = sweep(&models)?;
        select(&mut models, &mut saved, true);
        ensure!(sweep(&models)? == oracle);
        let warmup = Instant::now();
        let mut warmup_sweeps = 0;
        while warmup.elapsed().as_secs_f64() < 2. {
            for candidate in [false, true] {
                select(&mut models, &mut saved, candidate);
                ensure!(sweep(&models)? == oracle);
                warmup_sweeps += 1;
            }
        }
        let warmup_seconds = warmup.elapsed().as_secs_f64();
        let mut samples = Vec::new();
        for pair in 0..5 {
            for candidate in if pair % 2 == 0 {
                [false, true]
            } else {
                [true, false]
            } {
                select(&mut models, &mut saved, candidate);
                let start = Instant::now();
                let mut last = Vec::new();
                for _ in 0..32 {
                    last = sweep(&models)?;
                }
                let seconds = start.elapsed().as_secs_f64() / 32.;
                ensure!(seconds.is_finite() && seconds > 0. && last == oracle);
                samples
                    .push(serde_json::json!({"pair":pair,"candidate":candidate,"seconds":seconds}));
            }
        }
        ensure!(models.iter().all(|(moe, _)| moe.read_bytes == 0));
        std::fs::write(
            output,
            serde_json::to_vec_pretty(&serde_json::json!({
                "status":"complete","layers":48,"route_rows":2,"route_count":960,
                "resident_gate_bytes":48usize*512*H as usize*4,"warmup_sweeps":warmup_sweeps,
                "warmup_seconds":warmup_seconds,"sweeps_per_sample":32,"parity":"exact_route_ids",
                "no_expert_io":true,"samples":samples
            }))?,
        )?;
        println!(
            "FLASH_GATE_MICRO layers=48 rows=2 samples=10 sweeps=32 parity=exact_route_ids no_expert_io=true"
        );
        Ok(())
    }

                        input.try_clone()?
                    } else {
                        Array::concatenate(&[&input, &input], 0)?
                    };
                    let before = pool
                        .borrow()
                        .counters
                        .submitted
                        .load(std::sync::atomic::Ordering::Relaxed);
                    let output = current.forward_lookahead(&x, Some(&mut next), depth)?;
                    ensure!(
                        output.shape() == [rows, H]
                            && output.to_bytes()? == expected.repeat(rows as usize),
                        "lookahead changed MoE output"
                    );
                    let after = pool
                        .borrow()
                        .counters
                        .submitted
                        .load(std::sync::atomic::Ordering::Relaxed);
                    if depth > 0 && slots > 20 {
                        ensure!(after > before, "test did not submit predictions");
                    } else {
                        ensure!(
                            after == before,
                            "minimum cache must preserve demand capacity"
                        );
                    }
                    pool.borrow_mut().wait_for_test()?;
                    ensure!(
                        next.forward(&input)?.to_bytes()? == expected,
                        "prefetched weights differ"
                    );
                }
            }
            pool.borrow_mut().wait_for_test()?;
            pool.borrow_mut().poll();
            next.cache.borrow_mut().invalidate();
            let selected = next.route(&input)?.0.to_u32()?;
            let before = next.prefetched;
            next.prefetch(&selected, 4)?;
            pool.borrow_mut().wait_for_test()?;
            ensure!(next.forward(&input)?.to_bytes()? == expected);
            ensure!(
                if slots > 20 {
                    next.prefetched > before
                } else {
                    next.prefetched == before
                },
                "persistent admission or minimum-cache fallback was not exercised"
            );
            next.cache.borrow_mut().invalidate();
            let wrong = (0..512)
                .filter(|id| !selected.contains(id))
                .take(8)
                .collect::<Vec<_>>();
            ensure!(wrong.len() == 8);
            next.prefetch(&wrong, 4)?;
            let before = next.prefetched;
            pool.borrow_mut().wait_for_test()?;
            let ready = wrong
                .iter()
                .copied()
                .filter(|&id| next.cache.borrow().contains(id))
                .collect::<Vec<_>>();
            ensure!(ready.len() == if slots > 20 { 4 } else { 0 });
            ensure!(
                next.forward(&input)?.to_bytes()? == expected && next.prefetched == before,
                "wrong predictions changed real routing"
            );
            ensure!(
                ready.iter().all(|&id| next.cache.borrow().contains(id)),
                "unused records did not persist across a real forward"
            );
            let submitted = pool
                .borrow()
                .counters
                .submitted
                .load(std::sync::atomic::Ordering::Relaxed);
            if !ready.is_empty() {
                next.prefetch(&ready, 4)?;
                ensure!(
                    submitted
                        == pool
                            .borrow()
                            .counters
                            .submitted
                            .load(std::sync::atomic::Ordering::Relaxed),
                    "cached records were read twice"
                );
            }
            next.cache.borrow_mut().invalidate();
            next.prefetch(&[0, 511, 0], 4)?;
            let count = pool
                .borrow()
                .counters
                .submitted
                .load(std::sync::atomic::Ordering::Relaxed);
            next.prefetch(&[0, 511, 0], 4)?;
            ensure!(
                pool.borrow()
                    .counters
                    .submitted
                    .load(std::sync::atomic::Ordering::Relaxed)
                    == count,
                "duplicate speculative read"
            );
            pool.borrow_mut().wait_for_test()?;
            pool.borrow_mut().poll();
            ensure!(next.prefetch(&[512], 4).is_err() && next.prefetch(&[], 4).is_err());
            next.cache.borrow_mut().invalidate();
            let saved = next.records[0].data_offsets;
            next.records[0].data_offsets[0] = u64::MAX;
            next.prefetch(&[0, 511], 4)?;
            pool.borrow_mut().wait_for_test()?;
            ensure!(
                !next.cache.borrow_mut().reserve(&[0])?[0].1,
                "failed prediction left a false hit"
            );
            next.cache.borrow_mut().invalidate();
            next.records[0].data_offsets = saved;
            ensure!(
                next.forward(&input)?.to_bytes()? == expected,
                "prediction failure recovery differs"
            );
            ensure!(
                current
                    .forward_lookahead(&input, Some(&mut next), 11)
                    .is_err()
            );
            current.drain()?;
            next.drain()?;
        }
        Ok(())
    }

    #[test]
    #[ignore = "physical Apple GPU and independent Flash MoE inputs"]
    fn two_route_rows_match_oracle_with_cold_hot_small_caches_and_failed_reads()
    -> anyhow::Result<()> {
        use super::*;
        use serde_json::Value;
        use std::{fs::File, path::Path};
        let path = std::env::var("MLXL3_FLASH_REFERENCE")?;
        let root = Path::new(&path);
        let reference: Value = serde_json::from_reader(File::open(root.join("reference.json"))?)?;
        ensure!(
            reference["status"] == "complete" && reference["mode"] == "eval",
            "invalid MoE reference"
        );
        let checkpoint =
            crate::checkpoint::inspect(Path::new(&std::env::var("MLXL3_FLASH_MODEL")?))?;
        let mut inputs = Vec::new();
        let mut expected = Vec::new();
        for step in [0, 1, 4] {
            let entry = &reference["steps"][step]["components"][0]["mlp"];
            let read = |kind: &str| -> Result<Vec<u8>> {
                ensure!(entry[kind]["count"] == H, "invalid MoE fixture count");
                let bytes = std::fs::read(
                    root.join(entry[kind]["file"].as_str().context("missing fixture")?),
                )?;
                ensure!(
                    bytes.len() == H as usize * 4
                        && bytes
                            .as_chunks::<4>()
                            .0
                            .iter()
                            .all(|&b| f32::from_le_bytes(b).is_finite()),
                    "invalid MoE fixture bytes"
                );
                Ok(bytes)
            };
            inputs.push(Array::from_bytes(&read("input")?, &[1, H], Dtype::Float32)?);
            expected.push(read("output")?);
        }
        for slots in [10, 19, 20, 24, 53, 71, 80, 89, 512] {
            for (overlap, readers) in [
                (false, 1),
                (true, 1),
                (false, 2),
                (true, 2),
                (false, 4),
                (true, 4),
            ] {
                let mut moe = SsdMoe::load(
                    &checkpoint,
                    "model.language_model.layers.0.mlp",
                    slots,
                    overlap,
                )?;
                moe.readers = readers;
                moe.gpu_hot = true;
                moe.gpu_hot_wide = true;
                for pair in [[0, 1], [0, 1], [1, 0], [0, 0], [1, 2]] {
                    let x = Array::concatenate(&[&inputs[pair[0]], &inputs[pair[1]]], 0)?;
                    let output = moe.forward_many(&x)?.to_bytes()?;
                    let mut oracle = expected[pair[0]].clone();
                    oracle.extend(&expected[pair[1]]);
                    ensure!(
                        output == oracle,
                        "two-row MoE differs, slots={slots}, overlap={overlap}, pair={pair:?}"
                    );
                }
                for rows in 1..=8 {
                    let indices = (0..rows).map(|i| i % 3).collect::<Vec<_>>();
                    let x = Array::concatenate(
                        &indices.iter().map(|&i| &inputs[i]).collect::<Vec<_>>(),
                        0,
                    )?;
                    let oracle = indices
                        .iter()
                        .flat_map(|&i| expected[i].iter().copied())
                        .collect::<Vec<_>>();
                    for _ in 0..2 {
                        ensure!(
                            moe.forward_draft_rows(&x)?.to_bytes()? == oracle,
                            "wide MoE differs, slots={slots}, overlap={overlap}, rows={rows}"
                        );
                    }
                }
                let counters = (moe.hits, moe.misses, moe.read_bytes);
                for shape in [[0, H], [3, H], [2, H - 1]] {
                    ensure!(
                        moe.forward_many(&Array::zeros(&shape)?).is_err(),
                        "invalid route shape accepted"
                    );
                }
                ensure!(
                    moe.forward_many(&Array::zeros_dtype(&[1, H], Dtype::UInt32)?)
                        .is_err(),
                    "integer route input accepted"
                );
                for shape in [[0, H], [9, H], [4, H - 1]] {
                    ensure!(
                        moe.forward_draft_rows(&Array::zeros(&shape)?).is_err(),
                        "invalid draft route shape accepted"
                    );
                }
                ensure!(
                    counters == (moe.hits, moe.misses, moe.read_bytes),
                    "invalid shape changed cache counters"
                );
                moe.drain()?;
                moe.cache.borrow_mut().invalidate();
                let expert = moe.route(&inputs[0])?.0.to_u32()?[0] as usize;
                let original = moe.records[expert].data_offsets;
                moe.records[expert].data_offsets[0] = u64::MAX;
                let wide =
                    Array::concatenate(&[&inputs[0], &inputs[1], &inputs[0], &inputs[2]], 0)?;
                ensure!(
                    moe.forward_draft_rows(&wide).is_err(),
                    "failed source accepted"
                );
                let ids = moe.route(&inputs[0])?.0.to_u32()?;
                ensure!(
                    moe.cache.borrow_mut().reserve(&ids)?.iter().all(|p| !p.1),
                    "failed read left valid reservations"
                );
                moe.cache.borrow_mut().invalidate();
                moe.records[expert].data_offsets = original;
                ensure!(
                    moe.forward(&inputs[0])?.to_bytes()? == expected[0],
                    "retry after failed read diverged"
                );
                moe.drain()?;
            }
        }
        Ok(())
    }
    #[test]
    #[ignore = "physical Apple GPU; direct disk read into evaluated shared buffers"]
    fn disk_records_fill_only_owned_ranges_and_reject_bad_sources() -> anyhow::Result<()> {
        use crate::array::{Array, Dtype};
        use std::io::Write;
        let mut file = tempfile::tempfile()?;
        file.write_all(&(0..64u8).collect::<Vec<_>>())?;
        let mut banks = vec![
            Array::zeros_dtype(&[16], Dtype::UInt8)?,
            Array::zeros_dtype(&[16], Dtype::UInt8)?,
        ];
        Array::read_record(&file, 7, &mut banks, &[(0, 3, 5), (1, 9, 4), (0, 11, 2)])?;
        let mut a = vec![0u8; 16];
        a[3..8].copy_from_slice(&[7, 8, 9, 10, 11]);
        a[11..13].copy_from_slice(&[16, 17]);
        let mut b = vec![0u8; 16];
        b[9..13].copy_from_slice(&[12, 13, 14, 15]);
        assert_eq!(banks[0].to_bytes()?, a);
        assert_eq!(banks[1].to_bytes()?, b);
        for (offset, ranges) in [
            (0, vec![(2, 0, 1)]),
            (0, vec![(0, 16, 1)]),
            (0, vec![(0, usize::MAX, 1)]),
            (63, vec![(0, 0, 2)]),
            (u64::MAX, vec![(0, 0, 1)]),
            (0, vec![(0, 0, 0)]),
            (0, vec![]),
        ] {
            assert!(Array::read_record(&file, offset, &mut banks, &ranges).is_err());
            assert_eq!(banks[0].to_bytes()?, a);
            assert_eq!(banks[1].to_bytes()?, b);
        }
        Ok(())
    }

    #[test]
    #[ignore = "physical Apple GPU; concurrent disk reads into shared buffers"]
    fn disk_batches_reject_aliases_and_validate_every_record_before_writing() -> anyhow::Result<()>
    {
        use crate::array::{Array, Dtype};
        use std::io::Write;
        let mut file = tempfile::tempfile()?;
        file.write_all(&(0..64u8).collect::<Vec<_>>())?;
        for workers in 1..=4 {
            let mut banks = vec![Array::zeros_dtype(&[16], Dtype::UInt8)?];
            Array::read_records(
                &[(&file, 8), (&file, 20)],
                &mut banks,
                &[&[(0, 1, 4), (0, 5, 2)], &[(0, 9, 3), (0, 14, 2)]],
                workers,
            )?;
            let expected = vec![0, 8, 9, 10, 11, 12, 13, 0, 0, 20, 21, 22, 0, 0, 23, 24];
            assert_eq!(banks[0].to_bytes()?, expected);
            banks.push(banks[0].clone());
            for destinations in [
                vec![vec![(0, 0, 8)], vec![(0, 7, 2)]],
                vec![vec![(0, 0, 8)], vec![(1, 7, 2)]],
                vec![vec![(0, 0, 8)], vec![(0, 16, 2)]],
                vec![vec![(0, 0, 8)], vec![(2, 0, 2)]],
                vec![vec![(0, 0, 8)], vec![(0, 8, 2), (0, 10, 2)]],
            ] {
                let views = destinations.iter().map(Vec::as_slice).collect::<Vec<_>>();
                assert!(
                    Array::read_records(&[(&file, 0), (&file, 16)], &mut banks, &views, workers)
                        .is_err()
                );
                assert_eq!(banks[0].to_bytes()?, expected);
            }
            assert!(
                Array::read_records(
                    &[(&file, 0), (&file, 64)],
                    &mut banks,
                    &[&[(0, 0, 1)], &[(0, 1, 1)]],
                    workers
                )
                .is_err()
            );
            assert_eq!(banks[0].to_bytes()?, expected);
        }
        Ok(())
    }
