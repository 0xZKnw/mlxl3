import json,os,subprocess,time
from pathlib import Path
p=Path("docs/measurements/flash-next-ssd-impl/opt-66-fixed-width-real")
identity=json.loads((p/"identity-v1.json").read_text())
native=next(row["path"] for row in identity["artifacts"] if row["path"].endswith("mlxl3_native"))
name="qwen4::draft::runtime::tests::fixed_width_environment_reaches_the_production_option_reader"
result={"status":"incomplete","parity":False,"cases":[]}
try:
 for value,adaptive,expected in [("1","0","1"),("3","0","3"),("7","0","7"),(None,"0","3"),("0","0","error"),("8","0","error"),("01","0","error"),("+1","0","error"),(" 1","0","error"),("x","0","error"),("１","0","error"),("1","1","error")]:
  env=dict(os.environ,MLXL3_FLASH_DRAFT_ADAPTIVE=adaptive,MLXL3_EXPECTED_FIXED_WIDTH=expected)
  env.pop("MLXL3_FLASH_DRAFT_WIDTH",None)
  if value is not None:env["MLXL3_FLASH_DRAFT_WIDTH"]=value
  command=[native,name,"--exact","--ignored","--nocapture","--test-threads=1"]
  started=time.monotonic()
  child=subprocess.run(command,env=env,text=True,stdout=subprocess.PIPE,stderr=subprocess.STDOUT,timeout=10)
  (p/("host-options-"+str(len(result["cases"]))+".log")).write_text(child.stdout)
  row={"width":value,"adaptive":adaptive,"expected":expected,"exit":child.returncode,"seconds":time.monotonic()-started,"reaped":True}
  result["cases"].append(row)
  assert child.returncode==0 and "test result: ok. 1 passed; 0 failed;" in child.stdout
 result.update(status="passed",parity=True)
except BaseException as error:
 result.update(status="failed",parity=False,error=repr(error));raise
finally:
 (p/"host-options-result.json").write_text(json.dumps(result,indent=2)+"\n")
 print(result,flush=True)
