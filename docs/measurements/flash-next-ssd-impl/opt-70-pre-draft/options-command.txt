import hashlib,json,os,subprocess,time
from pathlib import Path
base=Path('docs/measurements/flash-next-ssd-impl/opt-70-pre-draft');identity=json.loads((base/'identity-v1.json').read_text())
native=next(r for r in identity['artifacts'] if r['path'].endswith('mlxl3_native'))
assert hashlib.sha256(Path(native['path']).read_bytes()).hexdigest()==native['sha256']
cases=[('default',None,None,1,'0'),('off','0','1',1,'0'),('on','1','1',1,'1'),('default-adaptive','1',None,1,'1'),('disabled','1','0',1,'error'),('fixed','0','0',1,'0'),('mode2','1','1',2,'0'),('mode3','1','1',3,'0'),('invalid-mode','1','1',4,'error')]
for n,v in [('two','2'),('true','true'),('empty',''),('space',' 1'),('negative','-1'),('nonunicode',os.fsdecode(b'\xff'))]:cases.append((n,v,'1',1,'error'))
result={'status':'incomplete','checks':[],'identity':'identity-v1.json'}
try:
 for name,value,adaptive,mode,expected in cases:
  env=os.environ.copy()
  for key,v in [('MLXL3_FLASH_DRAFT_PRECAP',value),('MLXL3_FLASH_DRAFT_ADAPTIVE',adaptive),('MLXL3_FLASH_DRAFT_WIDTH',None)]:
   env.pop(key,None)
   if v is not None:env[key]=v
  env.update(MLXL3_EXPECTED_PRECAP=expected,MLXL3_EXPECTED_MODE=str(mode))
  command=[native['path'],'qwen4::draft::runtime::tests::pre_draft_environment_reaches_real_options','--exact','--ignored','--nocapture','--test-threads=1']
  started=time.monotonic()
  with (base/f'options-{name}.log').open('x') as log:
   process=subprocess.run(command,env=env,stdout=log,stderr=subprocess.STDOUT,timeout=15)
  text=(base/f'options-{name}.log').read_text()
  row={'name':name,'expected':expected,'mode':mode,'exit':process.returncode,'seconds':time.monotonic()-started,'nonvacuous':'1 passed; 0 failed' in text};result['checks'].append(row)
  assert row['exit']==0 and row['nonvacuous'],row
 result['status']='passed_15_isolated_real_parser_checks'
except BaseException as error:result.update(status='failed_or_interrupted',error=repr(error));raise
finally:(base/'options-result.json').write_text(json.dumps(result,indent=2)+'\n');print(result,flush=True)
