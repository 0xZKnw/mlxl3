def hash_file(path):
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1 << 20), b""):
            digest.update(block)
    return digest.hexdigest()

    assert hashlib.sha256((w / path).read_bytes()).hexdigest() == h


def event_deltas(before, after):
    """Return nonnegative system event deltas in page-equivalent bytes."""
    pages = [re.search(r"page size of ([0-9]+) bytes", raw) for raw in [before, after]]
    if not all(pages) or int(pages[0][1]) <= 0 or pages[0][1] != pages[1][1]:
        raise ValueError("invalid or changed VM page size")
    result = {}
    for key in ["Pageins", "Compressions", "Decompressions", "Swapins", "Swapouts"]:
        values = [
            re.findall(rf"^{re.escape(key)}:\s+([0-9]+)\.\s*$", raw, re.MULTILINE)
            for raw in [before, after]
        ]
        if any(len(v) != 1 for v in values):
            raise ValueError("missing or duplicate VM event")
        delta = int(values[1][0]) - int(values[0][0])
        if delta < 0:
            raise ValueError("VM event counter moved backwards")
        result[key] = delta * int(pages[0][1])
    return result


def check_component(log):
    profiles = [
        json.loads(line.split("MTP43_COMPONENT ", 1)[1])
        for line in log.splitlines()
        if "MTP43_COMPONENT " in line
    ]
    assert len(profiles) == 1, "incomplete or duplicate component profile"
    profile = profiles[0]
    assert profile["status"] == "completed_original_components"
    for key, expected in [
        ("gate_layers", 48),
        ("gate_logit_count", 49152),
        ("head_logit_count", 496640),
        ("warmup_sweeps", 2),
        ("samples", 5),
        ("sweeps_per_sample", 16),
    ]:
        assert type(profile[key]) is int and profile[key] == expected
    for key in ["route_pair_48_layers_seconds", "head_pair_seconds"]:
        samples = profile[key]
        assert len(samples) == 5 and all(
            type(v) in [int, float] and math.isfinite(v) and v > 0 for v in samples
        )
    peak = profile["memory"]["process_lifetime_peak_bytes"]
    assert type(peak) is int and 0 < peak <= 10 << 30
    snapshots = [
        json.loads(line.split("MTP43_VM ", 1)[1])
        for line in log.splitlines()
        if "MTP43_VM " in line
    ]
    assert [s["phase"] for s in snapshots] == [
        "loaded",
        "inputs_ready",
        "warmup_done",
        "route_begin",
        "route_end",
        "head_begin",
        "head_end",
    ]
    assert all(
        type(s["seconds"]) in [int, float] and math.isfinite(s["seconds"]) and s["seconds"] >= 0
        for s in snapshots
    )
    assert all(b["seconds"] >= a["seconds"] for a, b in zip(snapshots, snapshots[1:]))
    events = [
        {
            "from": a["phase"],
            "to": b["phase"],
            "global_system_event_deltas_bytes": event_deltas(a["raw"], b["raw"]),
        }
        for a, b in zip(snapshots, snapshots[1:])
    ]
    return {"component": profile, "phase_system_event_deltas": events}


def check_head(log):
    profiles = [
        json.loads(line.split("MTP43_HEAD_PAIR ", 1)[1])
        for line in log.splitlines()
        if "MTP43_HEAD_PAIR " in line
    ]
    assert len(profiles) == 1, "missing or duplicate joint head profile"
    profile = profiles[0]
    assert profile["status"] == "passed_exact_head_components"
    for key, expected in [("checked_logits", 1986560), ("head_rows", 2), ("input_cases", 4)]:
        assert type(profile[key]) is int and profile[key] == expected
    blocks = profile["blocks"]
    assert len(blocks) == 5
    for i, block in enumerate(blocks):
        assert block["order"] == (["original", "joint"] if i % 2 == 0 else ["joint", "original"])
        for key in ["original", "joint"]:
            assert type(block[key]) in [int, float] and math.isfinite(block[key]) and block[key] > 0
    peak = profile["memory"]["process_lifetime_peak_bytes"]
    assert type(peak) is int and 0 < peak <= 10 << 30
    return {"head_components": profile}
