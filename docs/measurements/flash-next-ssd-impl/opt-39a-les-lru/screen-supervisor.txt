def choose(arms: list) -> int | None:
    if len(arms) != 8 or any(a.get("status") != "validated" for a in arms):
        return None
    gains = {d: [] for d in [2, 4]}
    for prompt in ["MAIN", "HOLD"]:
        cells = [a for a in arms if a["prompt"] == prompt]
        if sorted(a["depth"] for a in cells) != [0, 0, 2, 4]:
            return None
        if any(
            type(a["tps"]) not in [int, float] or not math.isfinite(a["tps"]) or a["tps"] <= 0
            for a in cells
        ):
            return None
        controls = [a for a in cells if a["depth"] == 0]
        if max(a["tps"] for a in controls) / min(a["tps"] for a in controls) > 1.10:
            return None
        baseline = max(a["tps"] for a in controls)
        peak = max(a["peak_bytes"] for a in controls)
        for depth in gains:
            c = next(a for a in cells if a["depth"] == depth)
            if c["peak_bytes"] > peak + (256 << 20):
                gains[depth].append(0)
            else:
                gains[depth].append(c["tps"] / baseline)
    eligible = [d for d, v in gains.items() if len(v) == 2 and all(x >= 1.01 for x in v)]
    return max(eligible, key=lambda d: (math.prod(gains[d]), -d)) if eligible else None


def power_ok(text: str, expected: str):
    if not text.splitlines() or text.splitlines()[0] != expected:
        raise ValueError("power changed during screen")
    if expected == "Now drawing from 'Battery Power'":
        import re

        match = re.search(r"\t(\d+)%;", text)
        if match is None or not 0 <= int(match[1]) <= 100:
            raise ValueError("invalid battery percentage")
        if int(match[1]) <= 5:
            raise KeyboardInterrupt("battery reserve <=5%")


def snapshot():
    data = {
        " ".join(c): subprocess.check_output(c, text=True, timeout=5)
        for c in [
            ["vm_stat"],
            ["sysctl", "vm.swapusage"],
            ["pmset", "-g", "batt"],
            ["pmset", "-g", "therm"],
        ]
    }
    data["vm_bytes"] = parse_vm(data["vm_stat"])
    return data
