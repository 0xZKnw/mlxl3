from pathlib import Path
import ast
import hashlib
import json
import math
import os
import runpy
import signal
import statistics
import subprocess
import time

w = Path.cwd()
root = Path("/Users/justin/Documents/mix-stq1_0")
p = w / "docs/measurements/flash-next-ssd-impl/opt-36-les-causal"
previous = p.parent / "opt-34-les-ready"
out = p / "release-controls"
out.mkdir(exist_ok=True)
assert not (out / "execution.json").exists()
identity = json.loads((previous / "identity.json").read_text())
for path, sha in identity["files"].items():
    assert hashlib.sha256((w / path).read_bytes()).hexdigest() == sha
for path, sha in identity["binaries"].items():
    assert hashlib.sha256(Path(path).read_bytes()).hexdigest() == sha
quality = json.loads((previous / "quality.json").read_text())
assert len(quality) == 24 and all(a["status"] == "passed" and a["reaped"] for a in quality)
source = ast.parse((previous / "screen-supervisor.txt").read_text())
driver = next(
    ast.literal_eval(n.value)
    for n in source.body
    if isinstance(n, ast.Assign)
    and any(isinstance(t, ast.Name) and t.id == "driver" for t in n.targets)
)
parse_vm = runpy.run_path(str(w / "benchmarks/benchmark_flash_memory.py"))["parse_vm"]
engines = {
    "H": root / "build/flash-opt-20/mixed-hc/mlxl3-rs",
    "C": root / "build/flash-opt-34/les-ready/mlxl3-rs",
}
assert (
    hashlib.sha256(engines["H"].read_bytes()).hexdigest()
    == "7a9a21205812dee34680df4b85600116b8ecb65c58d7897826206af54a8406a1"
)
plan = {
    "status": "diagnostic_only",
    "identity": identity,
    "modes": ["baseline", "mtp"],
    "schedule": [("H", 0), ("C", 0), ("C", 4), ("H", 0)],
    "tokens": 128,
    "warmup": 1,
    "repeats": 2,
    "prompt": "MAIN",
    "cache_gib": 5,
    "allocator_mib": 512,
    "readers": 4,
    "overlap": 1,
    "context_limit": 4096,
    "extra_peak_swap_bytes": 256 << 20,
    "within_arm_ratio": 1.25,
    "control_ratio": 1.10,
    "started_at": time.time(),
}
(out / "matrix.json").write_text(json.dumps(plan, indent=2) + "\n")


def snapshot():
    return {
        key: subprocess.check_output(cmd, text=True, timeout=5)
        for key, cmd in [
            ("power", ["pmset", "-g", "batt"]),
            ("vm", ["vm_stat"]),
            ("swap", ["sysctl", "vm.swapusage"]),
            ("thermal", ["pmset", "-g", "therm"]),
        ]
    }


def power(data):
    assert data["power"].splitlines()[0] == "Now drawing from 'AC Power'", "AC required"


records = []
result = {"status": "running", "parity_complete": False}
try:
    for mode in plan["modes"]:
        mtp_counters = None
        for index, (arm, depth) in enumerate(plan["schedule"]):
            name = f"{mode}-{index}-{arm}-depth{depth}"
            dest = out / name
            r = {
                "name": name,
                "mode": mode,
                "arm": arm,
                "depth": depth,
                "status": "running",
                "before": snapshot(),
            }
            power(r["before"])
            records.append(r)
            child = None
            started = time.monotonic()
            cmd = [
                str(root / ".venv/bin/python"),
                "-c",
                driver,
                str(engines[arm]),
                str(root / "models/Qwen3.8-Flash-Next-exl3-205-65c89531"),
                str(dest),
                mode,
                "MAIN",
                str(previous / "confirmation/reference128-MAIN.json"),
                str(p.parent / "opt-18-hc/speed/1-MAIN-baseline-H/report.json"),
            ]
            r["command"] = cmd
            r["env"] = {
                "MLXL3_SSD_CACHE_GIB": "5",
                "MLXL3_ALLOCATOR_CACHE_MIB": "512",
                "MLXL3_SSD_OVERLAP": "1",
                "MLXL3_SSD_READERS": "4",
                "MLXL3_SSD_LOOKAHEAD_DEPTH": str(depth),
            }
            try:
                with (out / (name + ".log")).open("w") as f:
                    child = subprocess.Popen(
                        cmd,
                        env={**os.environ, **r["env"]},
                        stdout=f,
                        stderr=subprocess.STDOUT,
                        start_new_session=True,
                    )
                    r["exit_code"] = child.wait(timeout=360)
                assert r["exit_code"] == 0, "bridge failed"
                report = json.loads((dest / "report.json").read_text())
                expected_sha = hashlib.sha256(engines[arm].read_bytes()).hexdigest()
                assert (
                    report["status"] == "validated"
                    and report["binary_sha256"] == expected_sha
                    and len(report["runs"]) == 2
                    and len(report["warmup"]) == 1
                )
                responses = report["warmup"] + report["runs"]
                oracle = json.loads((previous / "confirmation/reference128-MAIN.json").read_text())
                assert all(
                    x["token_hash"] == oracle["token_hash"] and x["text"] == oracle["cache_context"]
                    for x in responses
                )
                if mode == "mtp":
                    for x in responses:
                        counters = {k: v for k, v in x["stats"].items() if k.startswith("mtp_")}
                        if mtp_counters is None:
                            mtp_counters = counters
                        assert counters == mtp_counters, "MTP divergence"
                values = [x["stats"]["decode_tps"] for x in report["runs"]]
                assert all(type(x) in [int, float] and math.isfinite(x) and x > 0 for x in values)
                assert max(values) / min(values) <= 1.25, "within-arm drift"
                r["tps"] = statistics.median(values)
                r["peak_bytes"] = max(
                    x["stats"]["memory"]["process_lifetime_peak_bytes"] for x in responses
                )
                assert type(r["peak_bytes"]) is int and r["peak_bytes"] > 0
                r["after"] = snapshot()
                power(r["after"])
                r["swapout_bytes"] = (
                    parse_vm(r["after"]["vm"])["Swapouts"] - parse_vm(r["before"]["vm"])["Swapouts"]
                )
                assert 0 <= r["swapout_bytes"] <= 256 << 20, "swap guard"
                control_peaks = [
                    a["peak_bytes"]
                    for a in records
                    if a["mode"] == mode and a["arm"] == "H" and a["status"] == "validated"
                ]
                if arm == "C":
                    assert control_peaks and r["peak_bytes"] <= max(control_peaks) + (256 << 20), (
                        "extra process peak"
                    )
                r["status"] = "validated"
            except BaseException as exc:
                r.update(
                    status="interrupted" if isinstance(exc, KeyboardInterrupt) else "failed",
                    error=repr(exc),
                )
                raise
            finally:
                if child is not None and child.poll() is None:
                    os.killpg(child.pid, signal.SIGTERM)
                    try:
                        child.wait(timeout=5)
                    except subprocess.TimeoutExpired:
                        os.killpg(child.pid, signal.SIGKILL)
                        child.wait()
                r["reaped"] = child is None or child.poll() is not None
                r["seconds"] = time.monotonic() - started
                (out / "execution.json").write_text(json.dumps(records, indent=2) + "\n")
                with (w / "opti.md").open("a") as f:
                    f.write(
                        "\n- 36 contrôlecausalrelease "
                        + name
                        + " : **"
                        + r["status"]
                        + "**, "
                        + str(r.get("tps", "non mesuré"))
                        + "tok/s, swapouts+"
                        + str(r.get("swapout_bytes", "non mesuré"))
                        + "octets ; release-controls/execution.json ; diagnosticseulement.\n"
                    )
                print(
                    {k: r[k] for k in ["name", "status", "tps", "swapout_bytes"] if k in r},
                    flush=True,
                )
    rows = []
    assert len(records) == 8 and all(a["status"] == "validated" and a["reaped"] for a in records)
    for mode in plan["modes"]:
        cells = [a for a in records if a["mode"] == mode]
        controls = [a["tps"] for a in cells if a["arm"] == "H"]
        assert max(controls) / min(controls) <= 1.10, "control drift"
        zero = next(a for a in cells if a["arm"] == "C" and a["depth"] == 0)
        full = next(a for a in cells if a["arm"] == "C" and a["depth"] == 4)
        rows.append(
            {
                "mode": mode,
                "H_tps": controls,
                "C0_tps": zero["tps"],
                "C4_tps": full["tps"],
                "C0_vs_fastest_H_percent": 100 * (zero["tps"] / max(controls) - 1),
                "C4_vs_C0_percent": 100 * (full["tps"] / zero["tps"] - 1),
            }
        )
    result.update(status="diagnostic_complete", parity_complete=True, rows=rows)
except BaseException as exc:
    result.update(
        status="interrupted" if isinstance(exc, KeyboardInterrupt) else "failed", error=repr(exc)
    )
finally:
    result.update(
        completed_arms=sum(a["status"] == "validated" for a in records),
        planned_arms=8,
        not_started_arms=8 - len(records),
        finished_at=time.time(),
        limits="causal diagnostics, not confirmed gains; one C0/C4 cell per mode",
    )
    (out / "result.json").write_text(json.dumps(result, indent=2) + "\n")
    with (w / "opti.md").open("a") as f:
        f.write(
            "\n- **36 contrôlecausalrelease "
            + result["status"]
            + "** ; result.json, aucunevalidationdegain.\n"
        )
    print(result, flush=True)
